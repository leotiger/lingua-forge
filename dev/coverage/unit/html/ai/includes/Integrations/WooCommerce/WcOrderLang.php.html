216     * Hooked on `woocommerce_email_setup_locale` (filter on bool, WC 3+).
217     *
218     * WC only calls `setup_locale()` — and therefore reaches this filter — for
219     * `is_customer_email()` emails, so admin notifications are never touched.
220     *
221     * If a pending purchase language is available, we:
222     *  1. Switch the locale to that language's WP locale string.
223     *  2. Record the active lang in `$current_email_lang` (exposed via the
224     *     `linguaforge_email_order_lang` filter to WcPageBridge).
225     *  3. Return false so WC does NOT additionally switch to the customer's
226     *     saved user locale (which would override ours).
227     *
228     * @param  bool $setup WC's default: true (proceed with WC's own switch).
229     * @return bool        false when we handled the switch; $setup otherwise.
230     */
231    public static function maybe_switch_email_locale( bool $setup ): bool {
232        if ( '' === self::$pending_email_lang ) {
233            return $setup;
234        }
235
236        $lang   = self::$pending_email_lang;
237        $locale = Router::get_instance()->locale_from_lang( $lang );
238
239        switch_to_locale( $locale );
240
241        self::$locale_switched      = true;
242        self::$current_email_lang   = $lang;
243        self::$pending_email_lang   = '';
244
245        // Prevent WC from switching again (to the user's WP profile locale).
246        return false;
247    }
