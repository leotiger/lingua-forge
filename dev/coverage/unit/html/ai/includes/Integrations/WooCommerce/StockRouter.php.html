3 * Class LinguaForge\AI\Integrations\WooCommerce\StockRouter
4 *
5 * Routes WooCommerce stock WRITES to the source-language product.
6 *
7 * With the shared-stock delegation model, MetaDelegate transparently redirects
8 * all operational meta READS (including _stock, _stock_status, etc.) to the
9 * source product. However, WooCommerce stock WRITES — triggered by purchases,
10 * manual stock adjustments, and WC's stock management UI — target the product
11 * ID they were given. If a customer purchases a translated product, the order
12 * item carries the translated product ID, and WooCommerce will attempt to write
13 * the new stock level to that translated product post.
14 *
15 * Two complementary interception layers handle stock writes:
16 *
17 *  1. WordPress metadata filters (`update_post_metadata`, `add_post_metadata`)
18 *     intercept writes that go through `update_post_meta()` / `add_post_meta()`.
19 *     This covers the `$product->save()` path (WC calls `update_or_delete_post_meta()`
20 *     which calls `update_post_meta()`).
21 *
22 *  2. `woocommerce_update_product_stock_query` intercepts the *direct SQL* path
23 *     used by `WC_Product_Data_Store_CPT::update_product_stock()` and
24 *     `set_product_stock()`. These methods bypass the WordPress metadata API
25 *     entirely (`$wpdb->query("UPDATE postmeta SET meta_value…")`), so they are
26 *     invisible to layer 1. This filter rewrites the `post_id` in the SQL from
27 *     the translated product ID to the source product ID, and clears the source
28 *     product's postmeta object cache so that subsequent reads (including the
29 *     `update_lookup_table` call that follows in WC immediately) see the fresh
30 *     stock value.
31 *
32 * Stock keys covered by layer 1: _stock, _stock_status, _manage_stock,
33 * _backorders, _sold_individually.
34 *
35 * @package LinguaForge\AI\Integrations\WooCommerce
36 * @since   2.0.0
37 */
38

159    /**
160     * Rewrite the post_id in WooCommerce's direct SQL stock update to target
161     * the source product instead of the translated product.
162     *
163     * WooCommerce calls:
164     *   $sql = apply_filters( 'woocommerce_update_product_stock_query', $sql, $product_id, … )
165     *   $wpdb->query( $sql );
166     *   wp_cache_delete( $product_id, 'post_meta' );
167     *   $this->update_lookup_table( $product_id, 'wc_product_meta_lookup' );
168     *
169     * After this rewrite:
170     *  - The SQL's WHERE clause targets the source product → stock correctly decremented.
171     *  - $pending_cache_clear is populated so clear_source_meta_cache() can flush
172     *    the source product's postmeta object cache immediately after, ensuring that
173     *    the update_lookup_table() call that follows in WC reads the fresh stock value
174     *    for the translated product via MetaDelegate (which delegates to the source).
175     *
176     * @param  string         $sql        Raw SQL prepared by WC (post_id appears as a literal integer).
177     * @param  int            $product_id The translated (or source) product ID WC is targeting.
178     * @param  float|int|null $new_stock  The new stock quantity (passed by WC as the 3rd filter arg).
179     * @return string  Rewritten SQL, or original SQL when no routing is needed.
180     */
181    public static function rewrite_stock_sql( string $sql, int $product_id, float|int|null $new_stock = null ): string {
182
183        // ── 1. Post type guard ─────────────────────────────────────────────────
184        $post = get_post( $product_id );
185        if ( ! $post ) {
186            return $sql;
187        }
188
189        $delegate_types = (array) apply_filters( 'linguaforge_wc_delegate_post_types', [ 'product' ] );
190        if ( ! in_array( $post->post_type, $delegate_types, true ) ) {
191            return $sql;
192        }
193
194        // ── 2. Language guard — only rewrite for translated (non-source) posts ─
195        $lang = (string) get_post_meta( $product_id, '_lf_lang', true );
196        if ( '' === $lang ) {
197            return $sql;
198        }
199
200        $source_lang = Router::get_instance()->source_language();
201        if ( $lang === $source_lang ) {
202            return $sql;
203        }
204
205        // ── 3. Resolve source product ID ──────────────────────────────────────
206        $source_id = MetaDelegate::get_source_id_for( $product_id );
207        if ( ! $source_id || $source_id === $product_id ) {
208            return $sql; // Fail safe — let the write proceed on the translated post.
209        }
210
211        // ── 4. Rewrite post_id in the SQL ──────────────────────────────────────
212        // WC uses $wpdb->prepare() so the IDs appear as unquoted literal integers.
213        // Use a precise regex to avoid corrupting other integer literals in the SQL
214        // (e.g. WC_ROUNDING_PRECISION values, meta_value floats).
215        $rewritten = preg_replace(
216            '/\bpost_id\s*=\s*' . $product_id . '\b/',
217            'post_id = ' . $source_id,
218            $sql
219        );
220
221        if ( null === $rewritten || $rewritten === $sql ) {
222            // preg_replace failed or found nothing to replace — fail safe.
223            return $sql;
224        }
225
226        // ── 5. Schedule source cache clear ────────────────────────────────────
227        // WC calls wp_cache_delete($product_id) for the translated product after
228        // the SQL. We must also clear the source product's cache before WC's
229        // update_lookup_table() call reads postmeta for the translated product
230        // (MetaDelegate will delegate that read to the source, so the source
231        // cache must be fresh at that point).
232        // We clear it immediately here — the SQL has not run yet but the cache
233        // entry will be stale regardless once the SQL executes.
234        wp_cache_delete( $source_id, 'post_meta' );
235
236        // Record source_id + new_stock for clear_source_meta_cache() which:
237        //  (a) provides a second flush for any cache re-warmed between now and then, and
238        //  (b) syncs the source product's wc_product_meta_lookup row (WC-02).
239        self::$pending_cache_clear[ $product_id ] = [
240            'source_id' => $source_id,
241            'new_stock' => $new_stock,
242        ];
243
244        return $rewritten;
245    }

247    /**
248     * After WC's direct SQL stock write and update_lookup_table() complete:
249     *
250     *  1. Clear the source product's postmeta object cache (redundant flush for
251     *     any cache re-warmed since rewrite_stock_sql() ran).
252     *  2. Sync the source product's wc_product_meta_lookup row (WC-02).
253     *     WC's update_lookup_table() ran against the translated product ID and
254     *     correctly updated that row via MetaDelegate. But the source product's
255     *     own lookup row is stale — this update corrects it.
256     *
257     * Hooked on `woocommerce_updated_product_stock` at priority 1 (fires after
258     * WC's write and lookup update complete).
259     *
260     * @param int $product_id  The product ID WC targeted (the translated product).
261     */
262    public static function clear_source_meta_cache( int $product_id ): void {
263        global $wpdb;
264        /** @var \wpdb $wpdb */
265
266        $entry = self::$pending_cache_clear[ $product_id ] ?? null;
267        if ( ! $entry ) {
268            return;
269        }
270
271        $source_id = (int) $entry['source_id'];
272        $new_stock = $entry['new_stock'];
273        unset( self::$pending_cache_clear[ $product_id ] );
274
275        // ── 1. Flush postmeta object cache for source product ─────────────────
276        wp_cache_delete( $source_id, 'post_meta' );
277
278        // ── 2. Sync wc_product_meta_lookup for source product (WC-02) ─────────
279        // WC_Product_Data_Store_CPT::update_lookup_table() is a protected method
280        // with no public API equivalent. We write the two stock columns directly.
281        // $wpdb->wc_product_meta_lookup is registered by WC_Install::define_tables()
282        // on every request after plugins_loaded — guaranteed available here.
283        if ( empty( $wpdb->wc_product_meta_lookup ) ) {
284            return;
285        }
286
287        $manage_stock = get_post_meta( $source_id, '_manage_stock', true );
288
289        $wpdb->update( // phpcs:ignore WordPress.DB.DirectDatabaseQuery.DirectQuery,WordPress.DB.DirectDatabaseQuery.NoCaching -- WC_Data_Store_WP::update_lookup_table() is protected; no public API exists for this targeted two-column sync.
290            $wpdb->wc_product_meta_lookup,
291            [
292                'stock_quantity' => 'yes' === $manage_stock ? wc_stock_amount( $new_stock ) : null,
293                'stock_status'   => (string) get_post_meta( $source_id, '_stock_status', true ),
294            ],
295            [ 'product_id' => $source_id ],
296            [ '%f',         '%s' ],
297            [ '%d' ]
298        );
299
300        wp_cache_delete( 'lookup_table', 'object_' . $source_id );
301    }

307    /**
308     * Core routing logic shared by both update and add callbacks.
309     *
310     * @param  mixed  $check      Current filter value.
311     * @param  int    $object_id  Post ID receiving the write.
312     * @param  string $meta_key   Meta key.
313     * @param  mixed  $meta_value Value to write.
314     * @param  bool   $is_add     True for add_post_meta, false for update_post_meta.
315     * @return mixed
316     */
317    private static function maybe_route( mixed $check, int $object_id, string $meta_key, mixed $meta_value, bool $is_add ): mixed {
318
319        // ── 1. Key guard ───────────────────────────────────────────────────────
320        if ( ! in_array( $meta_key, self::STOCK_WRITE_KEYS, true ) ) {
321            return $check;
322        }
323
324        // ── 2. Reentrancy guard ────────────────────────────────────────────────
325        $guard = $object_id . ':' . $meta_key;
326        if ( isset( self::$routing[ $guard ] ) ) {
327            return $check;
328        }
329
330        // ── 3. Post type guard ─────────────────────────────────────────────────
331        $post = get_post( $object_id );
332        if ( ! $post ) {
333            return $check;
334        }
335
336        $delegate_types = (array) apply_filters( 'linguaforge_wc_delegate_post_types', [ 'product' ] );
337        if ( ! in_array( $post->post_type, $delegate_types, true ) ) {
338            return $check;
339        }
340
341        // ── 4. Language guard — only route for translated (non-source) posts ───
342        $lang = (string) get_post_meta( $object_id, '_lf_lang', true );
343        if ( '' === $lang ) {
344            return $check;
345        }
346
347        $source_lang = Router::get_instance()->source_language();
348        if ( $lang === $source_lang ) {
349            return $check; // Already writing to the source — no routing needed.
350        }
351
352        // ── 5. Resolve source product ID ──────────────────────────────────────
353        $source_id = MetaDelegate::get_source_id_for( $object_id );
354        if ( ! $source_id || $source_id === $object_id ) {
355            return $check; // Fail safe — let the write proceed on the translated post.
356        }
357
358        // ── 6. Route write to source product ──────────────────────────────────
