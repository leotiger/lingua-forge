2/**
3 * Class LinguaForge\AI\Integrations\WooCommerce\TermNameFilter
4 *
5 * Phase 1b: Translated term name display for WooCommerce taxonomies.
6 *
7 * In the shared-stock model, WooCommerce taxonomy terms (product categories,
8 * product tags, and product attribute values) are shared across all language
9 * versions of a product — there are no per-language term copies. Term names
10 * display in the source language by default.
11 *
12 * This class hooks into WordPress's `term_name` filter so that category and
13 * attribute names appear in the correct language for each visitor. Translated
14 * names are stored as termmeta under the key `_lf_term_name_{lang}` and are
15 * entered via the term edit screen (see TermNameAdmin).
16 *
17 * Covered taxonomies:
18 *   - product_cat   (WooCommerce product categories)
19 *   - product_tag   (WooCommerce product tags)
20 *   - product_type  (WooCommerce product type)
21 *   - pa_*          (WooCommerce product attribute value terms)
22 *
23 * The filter only fires when the current request language differs from the
24 * source language and a translated name is stored for the term + language
25 * combination. Otherwise the original name passes through unchanged.
26 *
27 * @package LinguaForge\AI\Integrations\WooCommerce
28 * @since   2.0.0
29 */

52        // a translated name exists for the detected request language.
53        //
54        // WordPress fires `term_name` from two distinct call sites with different
55        // signatures — we must accept 4 args to handle both:
56        //

72     * Replace the term name with its translated equivalent for WooCommerce
73     * taxonomies when the current request is in a non-source language and a
74     * translated name has been stored.
75     *
76     * Handles the two distinct WordPress call signatures for the `term_name`
77     * filter (see Trac #45085):
78     *
79     *   sanitize_term_field() path  → ($name, $term_id_int, $taxonomy, $context)
80     *   edit-tags.php path          → ($name, $wp_term_object)
81     *
82     * Only the 'display' context is translated; 'edit', 'db', and 'raw' are
83     * left unchanged so stored/raw values are never silently rewritten.
84     *

86     * @param  WP_Term|int|mixed $term_or_id  WP_Term (admin list) or int term ID (everywhere else).
87     * @param  string          $taxonomy    Taxonomy slug (set on sanitize_term_field path; empty on edit-tags path).
88     * @param  string          $context     Sanitize context: 'display', 'edit', 'db', 'raw', etc.
89     * @return string  Translated name if available, original name otherwise.
90     */
91    public static function translate_term_name( string $name, mixed $term_or_id, string $taxonomy = '', string $context = 'display' ): string {
92
93        // ── 1. Resolve term ID and taxonomy from whichever call signature fired ─
94        if ( $term_or_id instanceof WP_Term ) {
95            // edit-tags.php path: 2nd arg is a WP_Term object. Context is not passed,
96            // so it defaults to 'display' above — always a display context anyway.
97            $term_id  = $term_or_id->term_id;
98            $taxonomy = $term_or_id->taxonomy;
99        } elseif ( is_int( $term_or_id ) || ( is_string( $term_or_id ) && ctype_digit( (string) $term_or_id ) ) ) {
100            // sanitize_term_field() path: 2nd arg is the integer term ID.
101            $term_id = (int) $term_or_id;
102            // $taxonomy and $context are the 3rd and 4th args, already set above.
103        } else {
104            return $name;
105        }
106
107        // ── 2. Only translate for 'display' context ────────────────────────────
108        // 'edit', 'db', 'raw' contexts must return the stored source value.
109        if ( '' !== $context && 'display' !== $context ) {
110            return $name;
111        }
112
113        // ── 3. Only handle WooCommerce taxonomies ─────────────────────────────
114        if ( ! self::is_wc_taxonomy( $taxonomy ) ) {
115            return $name;
116        }
117
118        // ── 4. Resolve current request language ───────────────────────────────
119        $router = Router::get_instance();
120        $lang   = $router->detect_lang();
121
122        // ── 5. Skip if already in source language ─────────────────────────────
123        if ( '' === $lang || $lang === $router->source_language() ) {
124            return $name;
125        }
126
127        // ── 6. Look up translated name from termmeta ──────────────────────────
128        $translated = (string) get_term_meta( $term_id, self::META_PREFIX . $lang, true );
129
130        return '' !== $translated ? $translated : $name;
131    }

137    /**
138     * Returns true when $taxonomy is a WooCommerce taxonomy that participates
139     * in term name translation.
140     *
141     * Covers product_cat, product_tag, product_type, and all pa_* attribute
142     * taxonomies (WooCommerce registers one per product attribute).
143     */
144    public static function is_wc_taxonomy( string $taxonomy ): bool {
145
146        static $exact = [ 'product_cat', 'product_tag', 'product_type' ];
147
148        if ( in_array( $taxonomy, $exact, true ) ) {
149            return true;
150        }
151
