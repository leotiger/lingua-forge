3 * Class LinguaForge\AI\Integrations\WooCommerce\TermNameFilter
4 *
5 * Phase 1b: Translated term name display for WooCommerce taxonomies.
6 *
7 * In the shared-stock model, WooCommerce taxonomy terms (product categories,
8 * product tags, and product attribute values) are shared across all language
9 * versions of a product — there are no per-language term copies. Term names
10 * display in the source language by default.
11 *
12 * This class hooks into WordPress's `term_name` filter so that category and
13 * attribute names appear in the correct language for each visitor. Translated
14 * names are stored as termmeta under the key `_lf_term_name_{lang}` and are
15 * entered via the term edit screen (see TermNameAdmin).
16 *
17 * Covered taxonomies:
18 *   - product_cat   (WooCommerce product categories)
19 *   - product_tag   (WooCommerce product tags)
20 *   - product_type  (WooCommerce product type)
21 *   - pa_*          (WooCommerce product attribute value terms)
22 *
23 * The filter only fires when the current request language differs from the
24 * source language and a translated name is stored for the term + language
25 * combination. Otherwise the original name passes through unchanged.
26 *
27 * @package LinguaForge\AI\Integrations\WooCommerce
28 * @since   2.0.0
29 */
30

52        // a translated name exists for the detected request language.
53        //
54        // WordPress fires `term_name` from two distinct call sites with different
55        // signatures — we must accept 4 args to handle both:
56        //
57        //   1. sanitize_term_field()  → (string $name, int $term_id, string $taxonomy, string $context)

96    /**
97     * Replace the term name with its translated equivalent for WooCommerce
98     * taxonomies when the current request is in a non-source language and a
99     * translated name has been stored.
100     *
101     * Handles the two distinct WordPress call signatures for the `term_name`
102     * filter (see Trac #45085):
103     *
104     *   sanitize_term_field() path  → ($name, $term_id_int, $taxonomy, $context)
105     *   edit-tags.php path          → ($name, $wp_term_object)
106     *
107     * Only the 'display' context is translated; 'edit', 'db', and 'raw' are
108     * left unchanged so stored/raw values are never silently rewritten.
109     *
110     * @param  string          $name        The current term name.
111     * @param  WP_Term|int|mixed $term_or_id  WP_Term (admin list) or int term ID (everywhere else).
112     * @param  string          $taxonomy    Taxonomy slug (set on sanitize_term_field path; empty on edit-tags path).
113     * @param  string          $context     Sanitize context: 'display', 'edit', 'db', 'raw', etc.
114     * @return string  Translated name if available, original name otherwise.
115     */
116    public static function translate_term_name( string $name, mixed $term_or_id, string $taxonomy = '', string $context = 'display' ): string {
117
118        // ── 1. Resolve term ID and taxonomy from whichever call signature fired ─
119        if ( $term_or_id instanceof WP_Term ) {
120            // edit-tags.php path: 2nd arg is a WP_Term object. Context is not passed,
121            // so it defaults to 'display' above — always a display context anyway.
122            $term_id  = $term_or_id->term_id;
123            $taxonomy = $term_or_id->taxonomy;
124        } elseif ( is_int( $term_or_id ) || ( is_string( $term_or_id ) && ctype_digit( (string) $term_or_id ) ) ) {
125            // sanitize_term_field() path: 2nd arg is the integer term ID.
126            $term_id = (int) $term_or_id;
127            // $taxonomy and $context are the 3rd and 4th args, already set above.
128        } else {
129            return $name;
130        }
131
132        // ── 2. Only translate for 'display' context ────────────────────────────
133        // 'edit', 'db', 'raw' contexts must return the stored source value.
134        if ( '' !== $context && 'display' !== $context ) {
135            return $name;
136        }
137
138        // ── 3. Only handle WooCommerce taxonomies ─────────────────────────────
139        if ( ! self::is_wc_taxonomy( $taxonomy ) ) {
140            return $name;
141        }
142
143        // ── 4. Resolve current request language ───────────────────────────────
144        $router = Router::get_instance();
145        $lang   = $router->detect_lang();
146
147        // ── 5. Skip if already in source language ─────────────────────────────
148        if ( '' === $lang || $lang === $router->source_language() ) {
149            return $name;
150        }
151
152        // ── 6. Look up translated name from termmeta ──────────────────────────
153        $translated = (string) get_term_meta( $term_id, self::META_PREFIX . $lang, true );
154
155        return '' !== $translated ? $translated : $name;
156    }

190
191        if ( ! is_array( $terms ) || empty( $terms ) ) {
192            return $terms;
193        }
194
195        // ── 1. Resolve active language ─────────────────────────────────────────
196        $router      = Router::get_instance();
197        $source_lang = $router->source_language();
198
199        // Use _lf_lang from the queried product post (see translate_single_term_name).
200        //

202        // calling get_queried_object_id().  The latter calls
203        // WP_Query::get_queried_object() which, on taxonomy archive pages, calls

235            // the linguaforge_wc_delegate_taxonomies filter).
