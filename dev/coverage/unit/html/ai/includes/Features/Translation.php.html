298    /**
299     * Translation Memory fast path — block-level cache + batched API call.
300     *
301     * When TM is enabled and the post is TM-compatible (no block-attribute
302     * placeholders, known source language, standard post type), parse the
303     * content into top-level blocks, look each one up in the TM table, and
304     * send only the uncached blocks to the API as a batched request

310     * caller can fall back to the existing JSON-envelope flow.
311     *
312     * Cache key for each block:

317     * next translation. The compliance signature folds in the active
318     * compliance preset so toggling it on/off also invalidates.

623    /**
624     * Stable signature of the active preset's state at translation time.
625     *
626     * Folded into the Translation Memory cache key so that toggling the
627     * preset (or editing its temperature or per-preset addendum) invalidates
628     * affected cached translations — the new system prompt's stricter
629     * rules should actually apply on the next translation, not be hidden
630     * behind a stale cache.
631     *
632     * Reads via {@see Config::active_preset()} + {@see Config::preset_addendum()}
633     * so that:
634     *
635     *   • The 1.5.0+ per-preset addenda
636     *     (`linguaforge_preset_addendum_{technical,legal,creative}`) are
637     *     correctly reflected — editing any of them invalidates the cache
638     *     rows that used the previous value.
639     *   • A per-post `_linguaforge_preset` override participates in the
640     *     signature: a page that overrides to Legal gets its own cache
641     *     bucket distinct from the global-Technical bucket on the same
642     *     source markup.
643     *
644     * Note on backward compatibility: TM rows written before this fix
645     * (with the legacy `linguaforge_compliance_addendum`-derived signature)
646     * become permanent misses on first encounter. They are overwritten on
647     * the next translation rather than re-served, which is the correct
648     * outcome — those rows were keyed on a stale addendum.
649     *
650     * @param int $post_id Pass through so per-page preset overrides
651     *                     participate in the signature. 0 = global only.
652     */
653    private static function compute_compliance_signature( int $post_id = 0 ): string {
654
655        $preset   = Config::active_preset( $post_id );
656        $presets  = Config::presets();
657        $addendum = Config::preset_addendum( $preset );
658
659        return substr( hash( 'sha256',
660            ( $preset !== 'standard' ? '1' : '0' )
661            . '|' . (string) ( $presets[ $preset ]['temperature'] ?? 0.1 )
662            . '|' . md5( $addendum )
663        ), 0, 16 );
664    }
