198        if ( ! current_user_can( 'manage_options' ) ) {

204        $enabled = ! empty( $_POST['linguaforge_seo_og_enabled'] );
205        update_option( 'linguaforge_seo_og_enabled', $enabled ? 1 : 0, false );
206
207        $allowed_modes = [ 'auto', 'locale-only', 'full', 'disabled' ];
208        // phpcs:ignore WordPress.Security.ValidatedSanitizedInput.MissingUnslash -- sanitize_key normalises slashes.

210        if ( ! in_array( $mode, $allowed_modes, true ) ) {
211            $mode = 'auto';
212        }
213        update_option( 'linguaforge_seo_og_mode', $mode, false );
214
215        $default_image = isset( $_POST['linguaforge_seo_og_default_image'] )
216            ? esc_url_raw( wp_unslash( $_POST['linguaforge_seo_og_default_image'] ) )
217            : '';
218        update_option( 'linguaforge_seo_og_default_image', $default_image, false );

220        wp_safe_redirect( add_query_arg(
221            'lf_seo_og_saved',
222            '1',
