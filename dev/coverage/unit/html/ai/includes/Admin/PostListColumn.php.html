273    // Secondary-language Sync safeguards
274    //
275    // Two independent guards, each with its own setting, checked separately
276    // at every Sync call site (render_sync_button(), run_sync()):
277    //

279    //     only. Unchanged since 2.6.0; do not fold into the general guard
280    //     below — WooCommerce keeps its own dedicated toggle regardless of
281    //     the general one, since the primary product is WC's operational
282    //     source of truth (price, SKU, stock) and warrants a separate,
283    //     more visible opt-in.
284    //   • general_secondary_sync_blocked() — every OTHER post type. Added in
285    //     the same release to close the same hole for ordinary posts, pages,
286    //     and non-WC CPTs, which had no restriction at all before this.
287    //
288    // The two are mutually exclusive by post type (is_wc_product_post_type()
289    // routes a post to exactly one of them), so exactly one of the two ever
290    // applies to a given post — enabling one setting never implicitly
291    // enables the other.

539     *
540     * Replicates FillTranslationsCommand logic without the WP-CLI scaffolding.

542     * TRID-linked post or updates the existing one.
543     */
544    public static function ajax_fill_missing(): void {
545        // ── Auth + nonce ──────────────────────────────────────────────────────
546        check_ajax_referer( self::NONCE_NAME, 'nonce' );
547
548        $post_id = absint( $_POST['post_id'] ?? 0 ); // phpcs:ignore WordPress.Security.NonceVerification.Missing -- nonce verified by check_ajax_referer above.

551        }
552
553        if ( ! current_user_can( 'edit_post', $post_id ) ) {
554            wp_send_json_error( [ 'message' => 'Insufficient permissions.' ] );
555        }
556
557        $post = get_post( $post_id );
558        if ( ! $post instanceof \WP_Post ) {
559            wp_send_json_error( [ 'message' => 'Post not found.' ] );
560        }

562        // wp_navigation posts must be translated via the Router tab's FSE
563        // navigation pipeline (ajax_translate_fse_navigation), which applies
564        // label-only translation rules and URL rewriting. Reject here so a stale
565        // "Translate missing" button from before the TRID group was populated

567        if ( $post->post_type === 'wp_navigation' ) {
568            wp_send_json_error( [
569                'message' => __( 'Navigation posts are translated via the Router tab, not this button.', 'lingua-forge' ),
570            ] );
571        }
572
573        // ── Determine which languages are missing ─────────────────────────────
574        $router      = Router::get_instance();
575        $source_lang = (string) get_post_meta( $post_id, '_lf_lang', true );
576        if ( '' === $source_lang ) {
577            $source_lang = $router->source_language();
578        }
579
580        $all_langs    = $router->languages();
581        $translations = function_exists( 'linguaforge_get_translations' )
582            ? linguaforge_get_translations( $post_id )
583            : [];
584
585        $to_fill = [];
586        foreach ( $all_langs as $lang ) {
587            if ( $lang !== $source_lang && empty( $translations[ $lang ] ) ) {
588                $to_fill[] = $lang;
589            }
590        }
591
592        if ( empty( $to_fill ) ) {
593            wp_send_json_success( [
594                'message' => 'All translations already exist.',
595                'results' => [],
596            ] );
597        }

602        // from this filter prevents creating a broken product post that lacks
603        // price, stock, and other operational meta.

610                ),

632                'target_language' => $lang,
633                'translate_mode'  => 'full',

649                ? linguaforge_get_translations( $post_id )
650                : [];

654            } else {

656            }
657
658            if ( 'error' === $outcome['status'] ) {
659                ++$errors;
660            } else {

662                // created or updated.  Fired before meta-description generation so

668                // --with-meta-description flag.
669                self::generate_meta_description( (int) $outcome['id'] );
670            }
671
672            $results[ $lang ] = $outcome;
673        }
674
675        // ── Restore save hooks ────────────────────────────────────────────────
676        add_action( 'wp_after_insert_post', [ $router->sync,       'handle_save_post'   ], 10, 2 );
677        add_action( 'wp_after_insert_post', [ $router->trid_group, 'handle_cache_clear' ], 20 );
678
679        if ( $errors > 0 && $errors === count( $to_fill ) ) {
680            wp_send_json_error( [ 'message' => 'All translations failed.', 'results' => $results ] );
681        }
682
683        wp_send_json_success( [ 'results' => $results ] );
684    }
