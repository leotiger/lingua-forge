192    private function detect_lang_from_host( array $langs ): string {

360    /**
361     * Parse the HTTP Accept-Language header and return the best matching
362     * router-known language code, or an empty string when no match is found.
363     *
364     * The header value is already sanitized via sanitize_text_field() before
365     * parsing. Quality values default to 1.0 when absent (RFC 4647). Both
366     * exact two-char codes ('de') and regional tags ('de-DE', 'de-AT') are
367     * matched against the router's known two-char language list.
368     *
369     * @param  array<string> $langs  Known router language codes (e.g. ['ca','de','en']).
370     * @return string                Matched two-char language code, or '' on no match.
371     */
372    public function detect_browser_lang( array $langs ): string {
373        // phpcs:ignore WordPress.Security.ValidatedSanitizedInput.InputNotSanitized -- sanitized immediately via sanitize_text_field() below; value used only for string parsing against a whitelist.
374        $raw_header = isset( $_SERVER['HTTP_ACCEPT_LANGUAGE'] )
375            ? sanitize_text_field( wp_unslash( $_SERVER['HTTP_ACCEPT_LANGUAGE'] ) )
376            : '';
377
378        if ( $raw_header === '' ) {
379            return '';
380        }
381
382        // Parse "de-DE,de;q=0.9,en-US;q=0.8,en;q=0.7,ca;q=0.6".
383        $entries = [];
384        foreach ( array_map( 'trim', explode( ',', $raw_header ) ) as $entry ) {
385            $parts   = explode( ';', $entry );
386            $tag     = strtolower( trim( $parts[0] ) );
387            $quality = 1.0;
388            if ( isset( $parts[1] ) ) {
389                $qstr = trim( $parts[1] );
390                if ( str_starts_with( $qstr, 'q=' ) ) {
391                    $quality = (float) substr( $qstr, 2 );
392                }
393            }
394            $entries[] = [ 'tag' => $tag, 'q' => $quality ];
395        }
396
397        // Sort highest quality first.
398        usort( $entries, static fn( array $a, array $b ): int => $b['q'] <=> $a['q'] );
399
400        foreach ( $entries as $entry ) {
401            $tag = $entry['tag'];
402
403            // Exact two-char match (e.g. 'de').
404            if ( strlen( $tag ) === 2 && in_array( $tag, $langs, true ) ) {
405                return $tag;
406            }
407
408            // Prefix match for regional tags (e.g. 'de-de' → 'de', 'de-at' → 'de').
409            if ( strlen( $tag ) > 2 && $tag[2] === '-' ) {
410                $prefix = substr( $tag, 0, 2 );
411                if ( in_array( $prefix, $langs, true ) ) {
412                    return $prefix;
413                }
414            }
415        }
416
417        return '';
418    }
