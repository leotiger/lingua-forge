69    /**
70     * Derive a Lingua Forge "lang" code from a full WordPress locale string.
71     *
72     * Locales with an underscore (e.g. "de_DE", "zh_CN", "pt_PT_ao90") use the
73     * segment before the FIRST underscore — this is exactly the 2-character
74     * ISO 639-1 language subtag for every such locale WordPress ships.
75     *
76     * Locales with NO underscore are returned unchanged. Most of these are
77     * already a bare 2-character code ("ca", "ja", "th"), but WordPress's own
78     * locale registry also has roughly two dozen bare THREE-character codes
79     * for languages that have no ISO 639-1 code of their own — "yor" (Yoruba),
80     * "ckb" (Sorani Kurdish), "dsb" (Lower Sorbian), "sah" (Sakha), "arg"
81     * (Aragonese), and others.
82     *
83     * Before this method existed, every locale was unconditionally truncated
84     * to its first two characters (`substr($locale, 0, 2)`) regardless of
85     * length. That's harmless for a 3-character locale whose first two
86     * letters happen to coincide with its real ISO 639-1 code purely by
87     * chance (Yoruba's "yor" → "yo" does), but produces an outright wrong,
88     * different language's code for most of the others (e.g. "sah" (Sakha)
89     * truncates to "sa", which is Sanskrit's real code) — and even in the
90     * lucky "yor" → "yo" case, the *reverse* lookup in
91     * LocaleDetector::locale_from_lang('yo') can never find "yor" again,
92     * since it only matches a locale that IS the lang code or starts with
93     * "{lang}_", neither of which "yor" satisfies for lang "yo". Confirmed
94     * live: a site with the Yoruba language pack installed could never fully
95     * uninstall it — collect_locale_files() would find and delete "yor.mo",
96     * but "yo" (the truncated code) stayed in Context::languages() forever,
97     * because get_available_languages() kept returning "yor" as long as any
98     * OTHER trace of it existed, and even once none did, nothing ever
99     * resolved "yo" back to a real, loadable locale for the admin-bar
100     * preview switcher.
101     *
102     * This keeps WordPress's own locale slug verbatim for 3-letter-only
103     * locales rather than normalising it to the "textbook" ISO 639-1
104     * equivalent where one exists — internal routing/URLs/postmeta always
105     * use this value, and changing it after the fact would break already-
106     * published URLs and stored postmeta on any site already using it. For
107     * outbound-facing correctness (hreflang, og:locale, browser-language
108     * auto-detection, display labels) see iso_639_1_from_lang() below,
109     * which normalises just for those specific purposes without touching
110     * the internal lang code at all.
111     *
112     * @param  string $locale  A WordPress locale string (e.g. from
113     *                          get_locale() or get_available_languages()).
114     * @return string  Lowercased lang code.
115     */
116    public static function lang_from_locale( string $locale ): string {

118        $lang           = $underscore_pos !== false ? substr( $locale, 0, $underscore_pos ) : $locale;

122    /**
123     * Normalise a Lingua Forge lang code to its real ISO 639-1 equivalent,
124     * for outbound-facing uses only (hreflang tags, og:locale, ICU/CLDR
125     * display-name lookups, and matching a visitor's Accept-Language header).
126     *
127     * lang_from_locale() deliberately keeps WordPress's own locale slug
128     * verbatim for 3-letter-only locales, since changing the INTERNAL lang
129     * code after a site has already published URLs and stored postmeta with
130     * it would be a breaking change. But several outbound-facing systems do
131     * expect (or strongly prefer) the real ISO 639-1 code when one exists:
132     *
133     *   - BCP 47 primary subtags used in hreflang and og:locale are commonly
134     *     paired with a 2-letter region (e.g. "yo-NG"); WordPress's "yor"
135     *     slug produces a malformed tag if used directly.
136     *   - PHP's intl extension (backed by ICU/CLDR) generally does not
137     *     recognise WordPress's bare 3-letter slugs and silently echoes them
138     *     back unlabelled rather than producing a real display name.
139     *   - A visitor whose browser correctly reports `Accept-Language: yo`
140     *     (the real ISO code) won't match a router language of "yor" without
141     *     this reverse mapping.
142     *
143     * Only includes languages verified against the official ISO 639-1 ↔
144     * ISO 639-3 mapping (https://en.wikipedia.org/wiki/List_of_ISO_639_language_codes)
145     * where the two codes are unambiguous individual-language equivalents.
146     * Several of WordPress's other bare 3-letter locales are deliberately
147     * NOT included here because mapping them would be linguistically wrong,
148     * not just incomplete:
149     *   - "ckb" (Sorani Kurdish), "dsb"/"hsb" (Lower/Upper Sorbian), "sah"
150     *     (Sakha), "ceb" (Cebuano), "fur" (Friulian), "kab" (Kabyle), "haz"
151     *     (Hazaragi), "rhg" (Rohingya), "skr" (Saraiki), "szl" (Silesian) —
152     *     none of these have a real ISO 639-1 code at all.
153     *   - "ary" (Moroccan Arabic) and "azb" (South Azerbaijani) each have a
154     *     related macrolanguage code ("ar", "az") already used by WordPress
155     *     for a DIFFERENT, more general locale — mapping the dialect-specific
156     *     code onto the macrolanguage code would silently merge two distinct
157     *     languages together instead of correcting a naming mismatch.
158     *
159     * @param  string $lang  Lingua Forge lang code (from lang_from_locale()).
160     * @return string  The real ISO 639-1 code when a safe one exists, else
161     *                  $lang unchanged.

246     * Stored in the uploads folder so files survive plugin updates and are

252     * No code changes needed when adding new plugins or locales.

259        $upload = wp_upload_dir();

261        // phpcs:ignore WordPress.NamingConventions.PrefixAllGlobals.NonPrefixedHooknameFound -- lf_ is this plugin's registered short prefix; hook is public API.
262        return (string) apply_filters( 'lf_i18n_overrides_dir', $dir );

265    /**
266     * Scan the i18n-overrides/ directory and return every unique locale code
267     * found across all third-party .mo override files.
268     *
269     * Files follow the standard WordPress naming convention:
270     *   {textdomain}-{locale}.mo
271     *
272     * The locale suffix is a bare two- or three-letter code ("ca", "ja",
273     * "yor") or a language_COUNTRY pair ("it_IT", "pt_PT", "de_DE"). The
274     * three-letter case covers WordPress's own bare 3-character locale slugs
275     * for languages with no ISO 639-1 code of their own (see
276     * Context::lang_from_locale() for the full explanation) — every one of
277     * those in WordPress's locale registry is 3 characters, never more, so
278     * {2,3} covers the full range without over-matching. All forms are
279     * matched by the regex so any plugin's translation file is accepted
280     * automatically — no anchor on a specific text domain is needed.
281     *
282     * @return string[]  e.g. ['it_IT', 'pt_PT', 'de_DE', 'ca', 'yor', …]
283     */
284    public function discover_plugin_locales(): array {
285        $files   = glob( $this->i18n_overrides_dir() . '*.mo' ) ?: [];
286        $locales = [];
287
288        foreach ( $files as $file ) {
289            // Match the locale at the end: either "xx_XX" or bare "xx"/"xxx".
290            if ( preg_match( '/-([a-z]{2,3}(?:_[A-Z]{2})?)\.mo$/i', $file, $m ) ) {
291                $locales[] = $m[1];
292            }
293        }
294
295        return array_unique( $locales );
296    }

485     * exact two-char codes ('de') and regional tags ('de-DE', 'de-AT') are
486     * matched against the router's known two-char language list.

498            return '';
499        }

501        // Parse "de-DE,de;q=0.9,en-US;q=0.8,en;q=0.7,ca;q=0.6".
502        $entries = [];
503        foreach ( array_map( 'trim', explode( ',', $raw_header ) ) as $entry ) {
