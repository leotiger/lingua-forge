3 * Class LinguaForge\Router\Seo\SitemapManager
4 *
5 * Generates a dedicated multilingual XML sitemap at /lf-sitemap.xml that
6 * includes <xhtml:link rel="alternate" hreflang> entries for every
7 * translation group LF manages.
8 *
9 * ── Why a separate sitemap ────────────────────────────────────────────────
10 * The WordPress 5.5+ built-in sitemap has no hook to inject <xhtml:link>
11 * elements into individual <url> blocks — its XML renderer is closed.
12 * SEO plugin sitemaps (Yoast, Rank Math) replace the WP sitemap entirely
13 * but have no knowledge of LF's routing configuration.
14 *
15 * LF disables the WP core sitemap entirely (wp_sitemaps_enabled → false)
16 * and replaces it with /lf-sitemap.xml, which includes the hreflang
17 * alternates that search engines need to index multilingual content.
18 *
19 * Submit /lf-sitemap.xml to Google Search Console.
20 *
21 * ── Discovery ─────────────────────────────────────────────────────────────
22 * The sitemap URL is announced via a Sitemap: directive in robots.txt so
23 * crawlers find it automatically without manual submission.
24 *
25 * ── Caching ───────────────────────────────────────────────────────────────
26 * The sitemap is split into a sitemap index at /lf-sitemap.xml and per-chunk
27 * urlset files at /lf-sitemap-0.xml, /lf-sitemap-1.xml, …  Each chunk holds
28 * up to GROUPS_PER_CHUNK (1 000) TRID groups.  All chunks are generated in a
29 * single DB query and cached in separate transients (24 h TTL).  The index
30 * transient key is linguaforge_sitemap_xml; chunk keys are
31 * linguaforge_sitemap_chunk_{N}.  The cache is flushed automatically on any
32 * save_post that affects an LF-managed post and can be flushed manually from
33 * the admin panel.
34 *
35 * ── Options ───────────────────────────────────────────────────────────────
36 *   linguaforge_seo_sitemap_enabled  bool  Master switch (default true).
37 *
38 * ── Filters ───────────────────────────────────────────────────────────────
39 *   linguaforge_seo_sitemap_slug  string  URL slug (default 'lf-sitemap.xml').
40 *   linguaforge_seo_sitemap_xml   string  Sitemap index XML string before output.
41 *
42 * @package LinguaForge\Router\Seo
43 * @since   2.2.0
44 */

134    /**
135     * Send the headers for a sitemap (index or chunk) response.
136     *
137     * Split out from serve_xml() so it can be exercised in a test without
138     * triggering the exit() that immediately follows it in the real request
139     * flow — PHPUnit cannot observe assertions made after a test method hits
140     * exit(), so the header-emission behaviour would otherwise be untestable.
141     *
142     * status_header( 200 ) is required for chunk URLs, confirmed live against
143     * cal-talaia.cat: a chunk request (/lf-sitemap-0.xml) came back as an
144     * HTTP/2 404 with the correct body and content-type, and x-powered-by:
145     * PHP confirming this handler generated it. A chunk URL never matches a
146     * real post/page/rewrite rule, so WordPress's own request parsing has
147     * already called status_header( 404 ) before template_redirect fires —
148     * unlike robots.txt, which has a dedicated is_robots() fast-path that
149     * bypasses 404 determination entirely. The index URL (/lf-sitemap.xml)
150     * came back 200 on the same server without this fix, for reasons not yet
151     * confirmed (possibly a stale static file bypassing PHP rather than this
152     * code); status_header( 200 ) here is a safe no-op for that case either
153     * way, since it only overrides a status that hasn't been sent to the
154     * client yet. See IndexNowManager::send_key_file_headers() for the
155     * identical fix on the key-file route, where this was first found.
156     *
157     * A full-page cache or CDN in front of WordPress does not know this response
158     * is dynamic; without explicit no-cache headers it can freeze a stale hit for
159     * this exact URL (most commonly a 404 captured before any chunk existed, or a
160     * shrunk chunk count after posts were unpublished). Logged-in admin requests
161     * typically bypass such caches, so a manual browser check of a chunk URL can
162     * "load perfectly" while an anonymous crawler — including Googlebot fetching a
163     * <sitemap><loc> from the index — keeps receiving the stale response and never
164     * discovers the chunk's URLs. nocache_headers() forces every hit to run this
165     * handler fresh; the 24h transient cache (see class docblock) still avoids the
166     * expensive DB regeneration, so this doesn't reintroduce the query cost.

169        status_header( 200 );
170        nocache_headers();

174        // core's own defensive pattern and preventing a "headers already sent"
175        // PHP warning if anything upstream of template_redirect ever produces
176        // output first.
177        if ( ! headers_sent() ) {
178            header( 'Content-Type: application/xml; charset=UTF-8' );
179            header( 'X-Robots-Tag: noindex, follow' );

270    /**

274     * under a persistent object cache (Redis / Memcached) an individual chunk
275     * transient can be evicted independently while the index survives. Without a
276     * guard that would serve a valid-but-empty <urlset>, hiding every URL in the
277     * chunk from crawlers until the 24 h TTL lapses or a save flushes the cache.

279     * re-read it. The in-range check (against the stored chunk count) keeps an
280     * out-of-range or probe request from triggering a needless rebuild on every
281     * hit.

289
290        if ( ! is_string( $cached ) || '' === $cached ) {
291            $chunk_count = (int) get_option( 'linguaforge_sitemap_chunk_count', 0 );
292            if ( $chunk >= 0 && $chunk < $chunk_count ) {
293                $this->generate_and_cache();
294                $cached = get_transient( self::CACHE_KEY_CHUNK . $chunk );
295            }
296        }
297
298        return ( is_string( $cached ) && '' !== $cached ) ? $cached : $this->empty_urlset_xml();
299    }

323
324        // ── Fetch all published LF-managed posts ──────────────────────────────
