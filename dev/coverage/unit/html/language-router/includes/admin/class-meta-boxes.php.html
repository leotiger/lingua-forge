338            return;

460        // Strip all footnote markup from the imported content.

462        // internal block state — copying them verbatim breaks the block editor on
463        // the target page. The source footnotes are displayed in a read-only

467        $content = preg_replace( '/<!--\s*wp:footnotes\s*\/-->\n?/', '', $content );

469        //    the surrounding prose intact.

474            'post_title'   => $source->post_title,

477        ] );
