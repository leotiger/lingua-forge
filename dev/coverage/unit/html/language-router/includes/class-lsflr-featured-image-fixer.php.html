2/**
3 * Class LinguaForge\Router\FeaturedImageFixer
4 *
5 * Finds translated posts/pages/CPTs whose featured image is missing or
6 * out of sync with their source-language sibling, and lets an editor copy
7 * the source's featured image across — individually or in bulk.
8 *
9 * Why this exists: none of the built-in translation-creation paths
10 * (TranslationTrigger, AbstractTranslateCommand, PostListColumn's bulk
11 * "Translate" action) copy `_thumbnail_id` onto a newly created translation.
12 * A translated post is therefore born with no featured image unless an
13 * integration explicitly supplies one via the `linguaforge_translated_post_meta`
14 * filter. Translation-creation now copies the source's thumbnail going
15 * forward (see those three classes); this fixer covers translations that
16 * already existed before that change, plus any post whose thumbnail was
17 * changed on the source afterward and never re-synced on the translation.
18 *
19 * WooCommerce products are intentionally excluded: WooCommerce\MetaDelegate
20 * already serves `_thumbnail_id` from the source product at read time for
21 * every translation, so there is nothing to "fix" there — writing a copy
22 * would be silently shadowed by that delegation.
23 *
24 * UI: a "Fix Featured Images" button appears in the posts/pages/CPT list
25 * view whenever a language filter is active — same location and pattern as
26 * LinkFixer's "Fix Links" button. Clicking it opens a modal showing a
27 * dry-run scan, then lets the editor fix posts individually or all at once.
28 *
29 * Instantiated by the Router constructor as a sub-object
30 * ($router->featured_image_fixer).
31 */

123     * Compare one translated post's featured image against its source-language
124     * sibling in the same TRID group.
125     *
126     * Returns null when there is nothing to fix: no source translation exists,
127     * the source itself has no featured image, or the target's featured image
128     * already matches the source's.

131     *   post_id:       int,
132     *   title:         string,
133     *   current_id:    int,

137     *   source_url:    string
138     * }|null
139     */
140    public function scan_post( int $post_id, string $target_lang ): ?array {
141        $post = get_post( $post_id );
142        if ( ! $post ) return null;
143
144        $source_lang = $this->router->context->source_language();
145        if ( $target_lang === $source_lang ) return null; // nothing to compare the source against
146
147        $translations = $this->router->get_translations( $post_id );
148        $source_id    = (int) ( $translations[ $source_lang ] ?? 0 );
149        if ( ! $source_id || $source_id === $post_id ) return null;
150
151        $source_thumb = (int) get_post_thumbnail_id( $source_id );
152        if ( ! $source_thumb ) return null; // nothing to copy
153
154        $current_thumb = (int) get_post_thumbnail_id( $post_id );
155        if ( $current_thumb === $source_thumb ) return null; // already in sync
156
157        $current_url = $current_thumb ? (string) wp_get_attachment_image_url( $current_thumb, 'thumbnail' ) : '';
158        $source_url  = (string) wp_get_attachment_image_url( $source_thumb, 'thumbnail' );
159
160        return [
161            'post_id'      => $post_id,
162            'title'        => $post->post_title,
163            'current_id'   => $current_thumb,
164            'current_url'  => $current_url,
165            'source_id'    => $source_id,
166            'source_thumb' => $source_thumb,
167            'source_url'   => $source_url,
168        ];
169    }

171    /**
172     * Copy the source-language sibling's featured image onto a single post.
173     *
174     * Writes `_thumbnail_id` directly via update_post_meta() rather than calling
175     * WordPress's set_post_thumbnail(), which additionally requires
176     * wp_get_attachment_image() to successfully render the attachment before it
177     * will persist anything — a validity check meant for admin-supplied arbitrary
178     * attachment IDs. The ID copied here was already validated the moment it was
179     * set as the source post's own featured image, so re-validating it here would
180     * only add a redundant image-render lookup. This mirrors the direct
181     * meta_input writes the three translation-creation paths already use.
182     *
183     * @return array{ applied: bool }
184     */
185    public function fix_post( int $post_id, string $target_lang ): array {
186        $scan = $this->scan_post( $post_id, $target_lang );
187        if ( ! $scan ) return [ 'applied' => false ];
188
189        update_post_meta( $post_id, '_thumbnail_id', $scan['source_thumb'] );
190        $applied = ( (int) get_post_thumbnail_id( $post_id ) === $scan['source_thumb'] );
191
192        return [ 'applied' => $applied ];
193    }
