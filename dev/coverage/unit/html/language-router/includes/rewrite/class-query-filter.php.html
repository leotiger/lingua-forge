221    /**
222     * Adds a `_lf_lang` meta constraint to secondary (non-main) WP_Query
223     * instances on the public frontend.
224     *
225     * handle_pre_get_posts() guards with is_main_query() and therefore leaves
226     * all secondary queries unfiltered: sidebar widgets, `get_posts()` calls
227     * in templates, "Latest Posts" / "Latest Events" core blocks, and any
228     * other code that creates a WP_Query directly without going through the
229     * main query cycle.  Without this handler, those queries return results
230     * from all languages mixed together.
231     *
232     * WooCommerce post types are excluded — CatalogQuery already handles them
233     * via its own pre_get_posts hook and contains WC-specific logic (per-product
234     * `_lf_lang` override on language-neutral singular pages, related-products
235     * SQL patching, etc.) that must not be duplicated here.
236     *
237     * `post_type = 'any'` is skipped to avoid interfering with internal WP /
238     * WC queries that aggregate multiple post types, some of which may lack
239     * `_lf_lang` meta entirely.
240     *
241     * Third-party code can opt additional post types out of secondary filtering
242     * via the `linguaforge_secondary_query_excluded_post_types` filter.
243     *
244     * @param WP_Query $q  The secondary query being built.
245     */
246    public function handle_secondary_pre_get_posts( WP_Query $q ): void {
247        if ( $q->is_main_query() ) {
248            return;
249        }
250
251        if ( is_admin() ) {
252            return;
253        }
254
255        // REST API requests are not public-frontend queries. The link control,
256        // block renderer, and other editor REST calls have no URL language prefix,
257        // so LF_LANG would resolve to the source language and silently exclude all
258        // non-source-language content from search results. REST endpoints that need
259        // explicit language scoping use the lf_lang query param registered via
260        // filter_pages_by_lf_lang_rest() instead.
261        if ( defined( 'REST_REQUEST' ) && REST_REQUEST ) {
262            return;
263        }
264
265        if ( ! defined( 'LF_LANG' ) || '' === LF_LANG ) {
266            return;
267        }
268
269        $post_type = $q->get( 'post_type' );
270
271        // Skip 'any' — too broad; may include internal types without _lf_lang.
272        if ( 'any' === $post_type ) {
273            return;
274        }
275
276        // Skip ID-only lookups (fields='ids' or 'id=>parent'). These are
277        // internal infrastructure queries — not content displayed to users.
278        // WcPageBridge and similar helpers use get_posts(['fields'=>'ids'])
279        // to resolve translation page IDs; injecting a language constraint
280        // would exclude the very pages they are trying to find.
281        $fields = $q->get( 'fields' );

283            return;
284        }

286        // Normalize: empty string means WordPress will query 'post'.
287        $types = '' !== $post_type ? (array) $post_type : [ 'post' ];
288
289        // WordPress system / infrastructure post types — never carry _lf_lang meta
290        // and must not be language-filtered here. wp_navigation is especially
291        // critical: injecting a meta constraint causes WP_Navigation_Fallback to
292        // find zero results and create a brand-new navigation post from the latest
293        // classic menu, which manifests as unexpected navigation items on the
294        // frontend. nav_menu_item queries (classic menus) are also system-internal.
295        // This list mirrors internal_post_types() in class-sync.php (wp_navigation
296        // is intentionally included here, unlike the sync exclusion list).
297        $system_types = [
298            'wp_navigation', 'wp_navigation_fallback',
299            'nav_menu_item',
300            'wp_template', 'wp_template_part',
301            'wp_block', 'wp_global_styles',
302            'wp_font_family', 'wp_font_face',
303        ];
304        if ( array_intersect( $types, $system_types ) ) {
305            return;
306        }
307
308        // WC post types are handled by CatalogQuery — skip if any type in the
309        // query is a WC type to avoid double-injection or logic conflicts.
310        $wc_types = [ 'product', 'product_variation', 'shop_order', 'shop_coupon', 'shop_subscription', 'shop_booking' ];
311        if ( array_intersect( $types, $wc_types ) ) {
312            return;
313        }
314
315        // WordPress 6.3+ routes get_pages() through WP_Query, so pre_get_posts now
316        // fires for get_pages() calls made inside core/page-list during navigation
317        // rendering.  When the navigation arm is active (pending_page_list_lang is
318        // set), filter_page_list_frontend() handles language scoping for those
319        // get_pages() results.  Injecting a SQL meta_query here as well would
320        // filter out translated pages before filter_page_list_frontend sees them,

343            'key'   => '_lf_lang',

350    /**

352     * built-in known types and any types added via Settings → Router.

381    /**
382     * Filters get_pages() results to the active language on the public frontend
383     * and the Site Editor canvas. Also active in REST block-renderer requests
384     * when arm_page_list_lang_filter() has already set the pending language.
385     *
386     * If the language filter yields no pages (e.g. a translated WooCommerce product
387     * on a language-neutral URL whose nav pages have not yet been translated),
388     * the method falls back to showing source-language pages so the navigation
389     * is never left empty.
390     *
391     * Pages marked with _lf_page_menu_exclude are hidden in
392     * language filter so they appear in every language's navigation regardless
393     * of their own _lf_lang value.  Developers can extend the excluded-ID list
394     * via the linguaforge_page_menu_excluded_page_ids filter.
395     */
396    public function filter_page_list_frontend( array $pages, array $args ): array { // phpcs:ignore Generic.CodeAnalysis.UnusedFunctionParameter.FoundAfterLastUsed -- $args required by get_pages filter signature.
397        if ( defined( 'WP_CLI' ) && WP_CLI ) return $pages;
398        // Allow through when arm_page_list_lang_filter() has armed a language —
399        // this covers the Site Editor canvas (is_admin() + REST_REQUEST) where the
400        // navigation block renders server-side and core/page-list calls get_pages().
401        if ( $this->pending_page_list_lang === null ) {
402            if ( is_admin() ) {
403                return $pages; // admin context without a pending nav lang — nothing to filter
404            }
405            if ( defined( 'REST_REQUEST' ) && REST_REQUEST ) {
406                return $pages; // REST request without a pending nav lang — nothing to filter
407            }
408        }
409
410        if ( $this->pending_page_list_lang !== null ) {
411            $lang = $this->pending_page_list_lang;
412            // On the public frontend, block rendering is synchronous and
413            // clear_nav_lang_after_render() reliably fires after the navigation
414            // wrapper finishes — so keep pending alive for all get_pages() calls
415            // within that single navigation render (e.g. one call per sub-menu
416            // level or breakpoint variant).
417            // On canvas/REST (admin or REST_REQUEST), rendering can be interrupted
418            // mid-block, so consume immediately to prevent bleed into subsequent
419            // unrelated get_pages() calls.
420            if ( is_admin() || ( defined( 'REST_REQUEST' ) && REST_REQUEST ) ) {
421                $this->pending_page_list_lang = null;
422            }
423        } elseif ( defined( 'LF_LANG' ) ) {
424            // Frontend (non-admin, non-REST) — filter by the active site language.
425            $lang = LF_LANG;
426        } else {
427            return $pages;
428        }
429
430        $is_source = ( $lang === $this->router->context->source_language() );
431
432        // Collect IDs of pages explicitly excluded from all language navigations.
433        // We check only pages already in the get_pages() result — get_post_meta()
434        // is object-cached so this adds no DB overhead on warm requests.
435        $meta_excluded_ids = array_map(
436            fn( \WP_Post $p ) => $p->ID,
437            array_filter( $pages, fn( \WP_Post $p ) => (bool) get_post_meta( $p->ID, '_lf_page_menu_exclude', true ) )
438        );
439
440        /**
441         * Filters the page IDs that are hidden from every language's navigation
442         * page-list, regardless of their _lf_lang value.
443         *
444         * By default this list is derived from the _lf_page_menu_exclude postmeta
445         * flag set via the Language meta box or Quick Edit.  Developers can extend
446         * it programmatically — e.g. to always hide the privacy-policy page:
447         *
448         *     add_filter( 'linguaforge_page_menu_excluded_page_ids', function( $ids ) {
449         *         return array_merge( $ids, [ get_option( 'wp_page_for_privacy_policy' ) ] );
450         *     } );
451         *
452         * Note: this filter operates on core/page-list blocks inside core/navigation
453         * only.  Classic nav menus (wp_nav_menu) render from stored nav_menu_item
454         * posts and are unaffected.
455         *
456         * @param int[] $ids Page IDs to hide from every language navigation.
457         */
458        $excluded_ids = (array) apply_filters( 'linguaforge_page_menu_excluded_page_ids', $meta_excluded_ids ); // phpcs:ignore WordPress.NamingConventions.PrefixAllGlobals.NonPrefixedHooknameFound -- linguaforge_ is the registered plugin prefix.
459
460        $filtered = array_values( array_filter(
461            $pages,
462            function ( \WP_Post $page ) use ( $lang, $is_source, $excluded_ids ): bool {
463                if ( in_array( $page->ID, $excluded_ids, true ) ) return false;
464                $page_lang = get_post_meta( $page->ID, '_lf_lang', true );
465                if ( ! $page_lang ) return $is_source;
466                return $page_lang === $lang;
467            }
468        ) );
469
470        // Fallback for language-neutral URLs (e.g. WooCommerce product pages) where
471        // the queried post is a translation but no translated navigation pages exist
472        // yet.  Rather than rendering an empty navigation, show source-language pages
473        // so the menu is always present.  When translated nav pages are added later
474        // they will appear automatically without any extra configuration.
475        if ( ! $filtered && ! $is_source ) {
476            $source = $this->router->context->source_language();
477            $filtered = array_values( array_filter(
478                $pages,
479                function ( \WP_Post $page ) use ( $source, $excluded_ids ): bool {
480                    if ( in_array( $page->ID, $excluded_ids, true ) ) return false;
481                    $page_lang = get_post_meta( $page->ID, '_lf_lang', true );
482                    return ! $page_lang || $page_lang === $source;
483                }
484            ) );
485        }
486
487        return $filtered;
488    }

490    /**
491     * Arms the page-list lang filter for the Site Editor canvas and REST
492     * requests by reading the navigation block's language into
493     * $pending_page_list_lang before core/page-list calls get_pages().
494     *
495     * Handles two cases:
496     *  1. core/navigation block with a ref — wraps the page-list (template context).
497     *  2. core/page-list rendered directly inside a wp_navigation post — the post
498     *     content IS the page-list with no outer navigation wrapper (navigation
499     *     post editor in the Site Editor).
500     *
501     * Runs ONLY on the canvas (?canvas=edit) and REST API — never on regular
502     * admin or frontend requests.
503     */
504    public function arm_page_list_lang_filter( $pre_render, array $parsed_block, ?\WP_Block $parent_block = null ) { // phpcs:ignore Generic.CodeAnalysis.UnusedFunctionParameter.FoundAfterLastUsed -- $parent_block required by pre_render_block filter signature.
505        if ( defined( 'WP_CLI' ) && WP_CLI ) return $pre_render;
506
507        // phpcs:ignore WordPress.Security.NonceVerification.Recommended
508        $is_canvas = isset( $_GET['canvas'] ) && 'edit' === $_GET['canvas'];
509        $is_rest   = defined( 'REST_REQUEST' ) && REST_REQUEST;
510
511        $block_name = $parsed_block['blockName'] ?? '';
512
513        // ── Frontend path ─────────────────────────────────────────────────────────
514        // LF_LANG is URL-derived.  On prefixed URLs (/es/…, /ca/…) it is the right
515        // language.  On language-neutral URLs (/product/…) LF_LANG is always the
516        // source language, but the queried post may have a different _lf_lang —
517        // e.g. a Spanish product at /product/camisa/ has LF_LANG=en (source) even
518        // though its template uses header-es / navigation-es.
519        //
520        // Arm ONLY when all of the following hold:
521        //  1. We are on a language-neutral URL  (LF_LANG === source language).
522        //  2. The request is singular           (archives like /shop/ use LF_LANG).
523        //  3. The queried post has a non-source _lf_lang  (= it is a translation).
524        //  4. The navigation's _lf_lang matches that post language — the template
525        //     deliberately chose a language-specific nav for this post.
526        //
527        // In every other case filter_page_list_frontend() falls through to LF_LANG,
528        // which is correct for prefixed URLs, source-language singulars, and archives.
529        if ( ! $is_canvas && ! $is_rest ) {
530            // Arm when we are on a language-neutral URL (LF_LANG === source) but the
531            // queried post is a translation (post._lf_lang !== source).  In this case
532            // LF_LANG does not reflect the content language, so filter_page_list_frontend
533            // must be told which language to use rather than falling through to LF_LANG.
534            //
535            // We use the post's own _lf_lang as the pending language — not the
536            // navigation's language — so that the page-list shows pages in the correct
537            // language regardless of which navigation post the template chose.  This
538            // also handles CA/other languages when only an ES navigation exists: the
539            // page-list shows CA pages, not ES pages.
540            //
541            // The arm does NOT fire on prefixed URLs (LF_LANG !== source): there the
542            // fallback in filter_page_list_frontend already uses LF_LANG directly.
543            if ( $block_name === 'core/navigation' && defined( 'LF_LANG' ) ) {
544                $source = $this->router->context->source_language();
545                if ( LF_LANG === $source && is_singular() ) {
546                    $post = get_queried_object();
547                    if ( $post instanceof \WP_Post ) {
548                        $post_lang = (string) get_post_meta( $post->ID, '_lf_lang', true );
549                        if ( $post_lang && $post_lang !== $source ) {
550                            $this->pending_page_list_lang = $post_lang;
551                        }
552                    }
553                }
554            }
555            return $pre_render;
556        }
557
558        // ── Case 1: core/navigation wrapper (template context) ──────────────────
559        if ( $block_name === 'core/navigation' ) {
560            $ref = (int) ( $parsed_block['attrs']['ref'] ?? 0 );
561            if ( $ref <= 0 ) {
562                $fallbacks = get_posts( [
563                    'post_type'     => 'wp_navigation',
564                    'post_status'   => 'publish',
565                    'orderby'       => 'date',
566                    'order'         => 'DESC',
567                    'numberposts'   => 1,
568                    'no_found_rows' => true,
569                ] );
570                if ( empty( $fallbacks ) ) return $pre_render;
571                $ref = $fallbacks[0]->ID;
572            }
573
574            $this->pending_page_list_lang = $this->resolve_nav_lang( $ref );
575            return $pre_render;
576        }
577
578        // ── Case 2: core/page-list rendered directly as wp_navigation post content ─
579        // When editing a wp_navigation post in the Site Editor, the post content is
580        // bare <!-- wp:page-list /--> with no outer core/navigation wrapper, so the
581        // arm above never fires.  Detect the navigation post from the request context.
582        if ( $block_name === 'core/page-list' && $this->pending_page_list_lang === null ) {
583            $nav_id = $this->current_navigation_post_id();
584            if ( $nav_id > 0 ) {
585                $this->pending_page_list_lang = $this->resolve_nav_lang( $nav_id );
586            }
587        }
588
589        return $pre_render;
590    }
591
592    /**
593     * Returns the _lf_lang for a wp_navigation post, falling back to a
594     * -{lang} slug suffix, then to the source language.
595     */
596    private function resolve_nav_lang( int $nav_id ): string {
597        $lang = get_post_meta( $nav_id, '_lf_lang', true );
598        if ( ! $lang ) {
599            $post = get_post( $nav_id );
600            if ( $post ) {
601                foreach ( $this->router->context->languages() as $l ) {
602                    if ( str_ends_with( $post->post_name, '-' . $l ) ) {
603                        $lang = $l;
604                        break;
605                    }
606                }
607            }
608        }
609        return $lang ?: $this->router->context->source_language();
610    }
611
612    /**
613     * Detects the wp_navigation post being edited from the current request context.
614     * Covers both the canvas (?postType=wp_navigation&postId=N) and the REST
615     * block-renderer / navigation REST endpoint (?id=N or ?postId=N).
616     */
617    private function current_navigation_post_id(): int {
618        // Primary: parse the REST route — e.g. /wp/v2/navigation/62
619        // This is how the Site Editor fetches navigation posts for the canvas preview.
620        global $wp;
621        $rest_route = $wp->query_vars['rest_route'] ?? '';
622        if ( preg_match( '#^/wp/v2/navigation/(\d+)#', $rest_route, $m ) ) {
623            return (int) $m[1];
624        }
625
626        // Fallback: $_GET params (canvas URL context)
627        // phpcs:disable WordPress.Security.NonceVerification.Recommended
628        $candidates = [
629            (int) sanitize_text_field( wp_unslash( $_GET['postId']  ?? '' ) ),
630            (int) sanitize_text_field( wp_unslash( $_GET['post_id'] ?? '' ) ),
631            (int) sanitize_text_field( wp_unslash( $_GET['id']      ?? '' ) ),
632        ];
633        // phpcs:enable
634
635        foreach ( $candidates as $id ) {
636            if ( $id <= 0 ) continue;
637            $post = get_post( $id );
638            if ( $post && $post->post_type === 'wp_navigation' ) {
639                return $id;
640            }
641        }
642
643        // Last resort: global $post
644        global $post;
645        if ( $post instanceof \WP_Post && $post->post_type === 'wp_navigation' ) {
646            return $post->ID;
647        }
648
649        return 0;
650    }

652    /**
653     * Clears $pending_page_list_lang after the navigation block finishes
654     * rendering so it cannot bleed into subsequent get_pages() calls.
655     */
656    public function clear_nav_lang_after_render( string $block_content, array $parsed_block ): string {
657        $name = $parsed_block['blockName'] ?? '';
658        // Clear after the navigation wrapper (case 1) or after the bare page-list
659        // itself (case 2 — no wrapper, so we clear as soon as it finishes rendering).
660        if ( $name === 'core/navigation' || $name === 'core/page-list' ) {
661            $this->pending_page_list_lang = null;
662        }
663        return $block_content;
664    }

672            foreach ( $args['meta_query'] as $mq ) {
673                if ( isset( $mq['key'] ) && $mq['key'] === '_lf_lang' ) {
674                    return new WP_Query( $args );
675                }
676            }
677        }
678

685    public function query_fallback( array $args = [] ): WP_Query {
686        $lang   = defined( 'LF_LANG' ) ? LF_LANG : $this->router->context->source_language();
687        $source = $this->router->context->source_language();
688
689        $args['meta_query'][] = [
690            'relation' => 'OR',
691            [ 'key' => '_lf_lang', 'value' => $lang ],
692            [ 'key' => '_lf_lang', 'value' => $source ],
