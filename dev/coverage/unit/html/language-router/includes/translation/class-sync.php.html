127    public function assign_template_if_needed( int $post_id, $post, string $lang ): void {
128        $template_slug = $this->resolve_template_for_lang( $post, $lang );
129        $auto_prev     = (string) get_post_meta( $post_id, '_lf_auto_template', true );
130        $current       = (string) ( get_post_meta( $post_id, '_wp_page_template', true ) ?: '' );
131
132        // Back-compat / migration: _lf_auto_template is new in 1.3.3.
133        // If it has not been recorded yet but the current template matches the
134        // {base}-{lang} naming convention for any active language, treat it as
135        // a previously auto-assigned template so a language change can replace it.
136        if ( empty( $auto_prev ) && ! empty( $current ) && $current !== 'default' ) {
137            // Determine the template base name using the same logic as
138            // resolve_template_for_lang(): 'page', 'single', or 'single-{cpt}'.
139            if ( $post->post_type === 'page' ) {
140                $base = 'page';
141            } elseif ( $post->post_type === 'post' ) {
142                $base = 'single';
143            } else {
144                $base = 'single-' . $post->post_type;
145            }
146            foreach ( $this->router->context->languages() as $l ) {
147                if ( $current === $base . '-' . $l ) {
148                    $auto_prev = $current;
149                    break;
150                }
151            }
152        }
153
154        // Changing to the source language: revert _wp_page_template to 'default'
155        // only when the current template was auto-assigned by us — never touch a
156        // template the editor chose explicitly.
157        if ( ! $template_slug ) {
158            if ( $auto_prev && $current === $auto_prev ) {
159                update_post_meta( $post_id, '_wp_page_template', 'default' );
160                delete_post_meta( $post_id, '_lf_auto_template' );
161            }
162            return;
163        }
164
165        // No template_exists() guard here.  WordPress silently falls through to
166        // its normal template hierarchy when _wp_page_template holds a slug
167        // that is not registered, so assigning an as-yet-uncreated slug is
168        // harmless: the correct template will be used as soon as the theme
169        // file exists.  Any existence check that runs in the wp_after_insert_post
170        // context (file_exists, get_block_templates) can silently return false

172
173        // If the slug is already set to what we'd assign, just ensure the
174        // tracking key is recorded and return — avoids a redundant meta write.
175        if ( $current === $template_slug ) {
176            update_post_meta( $post_id, '_lf_auto_template', $template_slug );
177            return;
178        }
179
180        // Protect only explicitly user-chosen templates — i.e. non-empty,
181        // non-default, and NOT one we previously auto-assigned.  A previous
182        // auto-assignment (tracked in _lf_auto_template, or matching the
183        // {base}-{lang} naming convention for back-compat) is always replaceable
184        // when the language changes.
185        if ( ! empty( $current ) && $current !== 'default' && $current !== $auto_prev ) return;
186
187        update_post_meta( $post_id, '_wp_page_template', $template_slug );
188        update_post_meta( $post_id, '_lf_auto_template', $template_slug );
189    }

195    /**
196     * Prevents translated pages from being auto-added to navigation menus.
197     *
198     * WordPress's _wp_auto_add_pages_to_menus() fires on `publish_page` at
199     * priority 10 and adds new top-level pages to any classic nav menu that has
200     * "Automatically add new top-level pages" enabled. This is language-unaware:
201     * a newly published French page would be inserted into every auto-add menu,
202     * including the primary (source-language) menu.
203     *
204     * This callback runs at priority 11 — after the auto-add — and removes any
205     * item that references a non-source-language page. Source-language pages are
206     * allowed through normally; our translate_menu_items() filter in Redirector
207     * already handles dynamic URL swapping to translated permalinks at render time.
208     *
209     * @param int $post_id Published page ID.
210     */
211    public function remove_cross_language_menu_auto_add( int $post_id ): void {
212        $page_lang = get_post_meta( $post_id, '_lf_lang', true );
213        if ( ! $page_lang ) return;
214        if ( $page_lang === $this->router->context->source_language() ) return;
215
216        $nav_menu_options = get_option( 'nav_menu_options', [] );
217        if ( empty( $nav_menu_options['auto_add'] ) ) return;
218
219        foreach ( (array) $nav_menu_options['auto_add'] as $menu_id ) {
220            $menu_id = (int) $menu_id;
221            $items   = wp_get_nav_menu_items( $menu_id );
222            if ( ! $items ) continue;
223
224            $removed = false;
225            foreach ( $items as $item ) {
226                if ( $item->type === 'post_type' && (int) $item->object_id === $post_id ) {
227                    wp_delete_post( (int) $item->ID, true );
228                    $removed = true;
229                }
230            }
231
232            if ( $removed ) {
233                // wp_delete_post() clears the item's own post cache but not the
234                // nav menu items group cache keyed by menu term ID. Clear it
235                // explicitly so the removed item doesn't persist until cache expiry.
236                clean_term_cache( $menu_id, 'nav_menu' );
237            }
238        }
239    }

264    public function handle_save_post( int $post_id, $post ): void {
265        // wp_navigation needs language assignment even though it is not a public
266        // post type; all other types must be public and non-internal to qualify.
267        if ( $post->post_type !== 'wp_navigation' ) {
268            $pto = get_post_type_object( $post->post_type );
269            if ( ! $pto || ! $pto->public || in_array( $post->post_type, $this->internal_post_types(), true ) ) return;
270        }
271        if ( wp_is_post_revision( $post_id ) ) return;
272        if ( wp_is_post_autosave( $post_id ) ) return;
273        if ( ! current_user_can( 'edit_post', $post_id ) ) return;
274
275        // Dedicated metabox nonces. Verified independently of WP core's
276        // edit_post nonce so a CSRF on an unrelated POST endpoint that ends
277        // up triggering save_post cannot rebind language or translation groups.
278        //
279        // Missing-or-invalid nonce means "the field wasn't submitted by our
280        // metabox" — we silently skip the corresponding block rather than
281        // aborting the handler, so REST saves and wp_insert_post() callers
282        // (which post no nonce) continue to work for everything else.
283        $has_lang_nonce  = isset( $_POST['lf_language_nonce'] )
284            && wp_verify_nonce( sanitize_key( wp_unslash( $_POST['lf_language_nonce'] ) ), 'lf_language_save' );
285        $has_trans_nonce = isset( $_POST['lf_translations_nonce'] )
286            && wp_verify_nonce( sanitize_key( wp_unslash( $_POST['lf_translations_nonce'] ) ), 'lf_translations_save' );
287
288        $trid_group = $this->router->trid_group;
289        $context    = $this->router->context;
290
291        // Language
292        if ( $has_lang_nonce && isset( $_POST['lf_lang'] ) && $context->is_valid_lang( sanitize_key( wp_unslash( $_POST['lf_lang'] ) ) ) ) {
293            $trid_group->set_lang( $post_id, sanitize_key( wp_unslash( $_POST['lf_lang'] ) ) );
294        }
295        if ( ! get_post_meta( $post_id, '_lf_lang', true ) ) {
296            $trid_group->set_lang( $post_id, $context->source_language() );
297        }
298
299        $lang = $trid_group->get_lang( $post_id );
300
301        // Skip template/TRID/timestamp for wp_navigation and all non-content
302        // post types (internal WP types are excluded above; this gate is what
303        // keeps wp_navigation — public => false — out of this block).
304        $pto = get_post_type_object( $post->post_type );
305        if ( ! $pto || ! $pto->public || in_array( $post->post_type, $this->internal_post_types(), true ) ) return;
306
307        // ── Manual template selection — write BEFORE auto-assignment ──────────
308        // The lf_page_template POST field must be flushed to the DB before
309        // assign_template_if_needed() runs so that method reads the user's
310        // explicit choice as $current, not the stale DB value.
311        //
312        // Previous ordering had this block at the end of the function, which
313        // caused a hard overwrite: force_lang_template / assign_template_if_needed
314        // correctly assigned 'single-product-ca', then the POST block ran last
315        // and wrote 'default' back, winning on every classic-editor save.
316        $has_tpl_nonce = isset( $_POST['lf_template_nonce'] )
317            && wp_verify_nonce( sanitize_key( wp_unslash( $_POST['lf_template_nonce'] ) ), 'lf_template_save' );
318        if ( $has_tpl_nonce && isset( $_POST['lf_page_template'] ) ) {
319            update_post_meta(
320                $post_id,
321                '_wp_page_template',
322                sanitize_text_field( wp_unslash( $_POST['lf_page_template'] ) )
323            );
324        }
325
326        // Template auto-assignment.
327        //
328        // Fires whenever the post's language has changed OR when this is the
329        // first save the post has ever seen ($previous_lang empty). The
330        // in-method guard inside assign_template_if_needed() leaves any
331        // explicit admin template choice intact — only acts when the
332        // current template is 'default' or empty — so this is safe to
333        // trigger more aggressively than the original "on change only" gate.
334        //
335        // Without the first-save branch, posts created programmatically (REST
336        // inserts, wp_insert_post calls, duplicated-from-source flows) with
337        // _lf_lang already set would never get their language-specific template
338        // assigned, since there's no _lf_lang_previous to compare against.
339        // Template auto-assignment runs on every save — not just on language
