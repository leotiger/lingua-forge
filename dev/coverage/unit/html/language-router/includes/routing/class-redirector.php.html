304                if ( $post_lang && $post_lang !== $source ) {

308        }

311
312        if ( $front_id > 0 ) {

421     * Redirects a base template-part lookup to its per-language variant.
422     *
423     * WordPress resolves template parts via get_block_template( "{theme}//{slug}", 'wp_template_part' ).
424     * Theme-provided archive templates (WooCommerce shop, product_cat / product_tag,
425     * 404, etc.) hardcode the base slug (e.g. "footer") in their block markup.
426     * LF-assigned per-post templates already reference "footer-{lang}" directly
427     * and are unaffected.
428     *
429     * When this filter fires for a base slug:
430     *   1. Guards exit early if LF_LANG is not set, equals the source language,
431     *      if the theme prefix does not match the active stylesheet (plugin-owned
432     *      parts such as "woocommerce//checkout-header" are never redirected —
433     *      no "-{lang}" variant exists on the plugin filesystem), or if the slug
434     *      already ends with "-{lang}" (reentrancy / double-suffix).
435     *   2. A lang-specific variant is looked up: "{theme}//{slug}-{lang}".
436     *   3. If found, it is returned instead of the base template part.
437     *
438     * @param  \WP_Block_Template|null $template       Resolved template (or null).
439     * @param  string                  $id             Template identifier: "{theme}//{slug}".
440     * @param  string                  $template_type  'wp_template' or 'wp_template_part'.
441     * @return \WP_Block_Template|null
442     */
443    public function redirect_template_part_to_lang( $template, string $id, string $template_type ) {
444        // Only act on template parts — templates (page-level) are handled by Search\Query
445        // and FrontPageQuery which swap entire templates via get_block_templates.
446        if ( 'wp_template_part' !== $template_type ) {
447            return $template;
448        }
449
450        if ( is_admin() ) {
451            return $template;
452        }
453
454        if ( ! defined( 'LF_LANG' ) || '' === LF_LANG ) {
455            return $template;
456        }
457
458        if ( LF_LANG === $this->router->context->source_language() ) {
459            return $template;
460        }
461
462        // Parse "{theme}//{slug}".
463        $sep = strpos( $id, '//' );
464        if ( false === $sep ) {
465            return $template;
466        }
467
468        // Only redirect template parts that belong to the active theme.
469        // Parts owned by plugins (e.g. "woocommerce//checkout-header") must never
470        // be redirected — they live on the plugin filesystem and no "-{lang}" variant
471        // exists there, causing a PHP file_get_contents warning in WC's BlockTemplateUtils.
472        $theme_prefix = substr( $id, 0, $sep );
473        if ( $theme_prefix !== get_stylesheet() ) {
474            return $template;
475        }
476
477        $slug = substr( $id, $sep + 2 );
478
479        // Guard: slug already carries the lang suffix.
480        // • Ordinary path: the part slug was already localised (either by the
481        //   redirect_template_part_via_block_data hook above, or because the page
482        //   template references "footer-{lang}" directly).
483        // • Reentrancy: the get_block_template() call on line ~415 below re-enters
484        //   this filter for the localised slug; in both cases we must not recurse.
485        //
486        // When $template is still null WordPress will run its own DB query, which
487        // requires a matching wp_theme taxonomy term.  Fall back to a slug-only
488        // get_block_templates() lookup so the part is found even when its wp_theme
489        // term is absent or wrong (e.g. before RepairHandler has been run).
490        if ( str_ends_with( $slug, '-' . LF_LANG ) ) {
491            if ( null !== $template ) {
492                return $template; // already resolved by an earlier filter
493            }
494            $candidates = get_block_templates( [ 'slug__in' => [ $slug ] ], 'wp_template_part' );
495            return ! empty( $candidates ) ? $candidates[0] : $template;
496        }
497
498        // Look up the lang-specific variant.  The recursive call above will trip the
499        // str_ends_with guard and return immediately, so there is no infinite loop.
500        $lang_template = get_block_template( substr( $id, 0, $sep + 2 ) . $slug . '-' . LF_LANG, 'wp_template_part' );
501
502        return ( $lang_template instanceof \WP_Block_Template ) ? $lang_template : $template;
503    }
