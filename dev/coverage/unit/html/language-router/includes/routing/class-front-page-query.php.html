3 * Class LinguaForge\Router\Routing\FrontPageQuery
4 *
5 * Substitutes the active-language front-page block template at runtime.
6 *
7 * When a block theme ships a `front-page.html` template, WordPress applies it
8 * automatically whenever the site's static front page is rendered. This class
9 * mirrors the pattern used in Search\Query::override_search_template(): it hooks
10 * into `get_block_templates` on the frontend and, when `is_front_page()` is true
11 * and a language-specific variant exists (e.g. `front-page-es`), substitutes it
12 * for the base `front-page` template so the correct localised layout is used.
13 *
14 * The hook is a no-op when:
15 *   - LF_LANG is not defined (frontend without a language prefix, or admin/REST).
16 *   - `is_front_page()` returns false.
17 *   - No `front-page-{lang}` template exists in the DB or theme files.
18 * In all those cases the original `$templates` array is returned unchanged.
19 */
20
21namespace LinguaForge\Router\Routing;

25class FrontPageQuery {

28     * Reentrancy guard — prevents infinite recursion when we call get_block_templates() inside the filter.
29     *
30     * @var bool
31     */

33
34    public function register_hooks(): void {
35        add_filter( 'get_block_templates', [ $this, 'override_front_page_template' ], 10, 3 );
36    }

38    /**
39     * Swap `front-page` template for `front-page-{lang}` on the frontend.
40     *
41     * Only acts when:
42     *   - We are resolving wp_template objects (not wp_template_part).
43     *   - LF_LANG is defined and non-empty (i.e. a language prefix is active).
44     *   - is_front_page() is true (static front page or blog index as front page).
45     *   - A `front-page-{lang}` template actually exists (DB or theme file).
46     *
47     * @param  \WP_Block_Template[] $templates     Current template candidates.
48     * @param  array                $query         Original query args passed to get_block_templates().
49     * @param  string               $template_type 'wp_template' or 'wp_template_part'.
50     * @return \WP_Block_Template[]
51     */
52    public function override_front_page_template( array $templates, $query, string $template_type ): array {
53        if ( $template_type !== 'wp_template' ) return $templates;
54        if ( ! defined( 'LF_LANG' ) || LF_LANG === '' ) return $templates;
55        if ( ! is_front_page() ) return $templates;
56        if ( $this->in_override ) return $templates;
57
58        $this->in_override = true;
59        $lang_slug         = 'front-page-' . LF_LANG;
60        $found             = get_block_templates( [ 'slug__in' => [ $lang_slug ] ] );
61        $this->in_override = false;
62
63        return ! empty( $found ) ? $found : $templates;
64    }
