3 * Class LinguaForge\AI\Integrations\WooCommerce\AdminSaveGuard
4 *
5 * Controls translated WooCommerce product saves from the admin editor.
6 *
7 * Intercepts at save_post priority 0 to: (1) allow post-row writes (title,

15 * flag_source_sku_conflict() for the two same-named-but-opposite-polarity SKU
16 * uniqueness filters this class hooks.

105     * Detect an admin-form save of a translated product and configure the
106     * request for text-only persistence.
107     *
108     * Fires on save_post at priority 0 — before WooCommerce's meta-box save at
109     * priority 1 (WC_Admin_Meta_Boxes::save_meta_boxes).  save_post fires before
110     * save_post_{post_type}, so hooking save_post_product would be too late.
111     *
112     * If the post is a 'product', is a translated (non-source) product, and the
113     * WooCommerce meta nonce is present in $_POST (i.e. this is a direct
114     * admin-editor save, not a programmatic call), the method:
115     *
116     *   • Sets $intercepting = true so whitelist_meta_write() activates.
117     *   • Disables VariationSync for this request.
118     *   • Registers teardown_intercept() on redirect_post_location (priority 99)
119     *     so state is restored after edit_post() has fully returned.
120     *
121     * @param int      $post_id Product post ID.
122     * @param \WP_Post $post    Post object.
123     */
124    public static function maybe_intercept_translated_save( int $post_id, \WP_Post $post ): void {
125        // save_post fires for every post type; bail early for non-products.
126        if ( 'product' !== $post->post_type ) {
127            return;
128        }
129
130        if ( self::$intercepting ) {
131            return; // Re-entrant call from WC's $variable_product->save() — already set.
132        }
133
134        // Only intercept direct admin-editor saves (WC nonce present).
135        // phpcs:ignore WordPress.Security.NonceVerification.Missing -- existence check only; verification is WC's responsibility.
136        if ( empty( $_POST['woocommerce_meta_nonce'] ) ) {
137            return;
138        }
139
140        $lang = (string) get_post_meta( $post_id, '_lf_lang', true );
141        if ( '' === $lang ) {
142            return; // Not LF-managed.
143        }
144
145        if ( $lang === Router::get_instance()->source_language() ) {
146            return; // Source product — normal WC save; SKU notice handled by filter_sku_error_on_display().
147        }
148
149        self::$intercepting = true;
150
151        // Remove identity keys from $_POST before WC's save_meta_boxes (priority 1) runs.
152        //
153        // WC_Product::set_sku() and set_global_unique_id() both unconditionally call
154        // a direct SQL uniqueness query (wc_product_has_unique_sku /
155        // wc_product_has_unique_global_unique_id).  Those queries find the source
156        // product's row (same identifier) and return $is_unique = false.  WC then
157        // calls $this->error() which throws WC_Data_Exception.  On some hosts this
158        // exception is not caught by WC_Admin_Meta_Boxes::save_meta_boxes(),
159        // propagates through the save_post dispatcher, and terminates the request
160        // (blank page / PHP timeout).
161        //
162        // Both keys are owned by MetaDelegate (delegated from the source product at
163        // runtime) so there is nothing to validate or write during a text-only admin
164        // save.  Unsetting them causes WC_Meta_Box_Product_Data::save() to skip the
165        // isset() branches entirely — no setter call, no SQL, no exception.
166        // phpcs:ignore WordPress.Security.NonceVerification.Missing
167        unset( $_POST['_sku'], $_POST['_global_unique_id'] );
168
169        // Variation sync is not needed on admin saves: retranslation builds the
170        // full structure; running it here would cascade into WC's price hooks.
171        // Stays disabled until teardown_intercept() fires at redirect_post_location.
172        VariationSync::disable_for_request();
173
174        // Tear down only AFTER edit_post() has fully returned (all inner and outer
175        // wp_after_insert_post calls complete).  redirect_post_location is the
176        // earliest reliable hook that fires after the entire save chain, and it
177        // only fires in the classic admin path — the same path that presents the
178        // woocommerce_meta_nonce we checked above.
179        add_filter( 'redirect_post_location', [ self::class, 'teardown_intercept' ], 99 );
180    }

182    /**
183     * Allow only whitelisted meta keys to be written during an intercepted
184     * translated-product admin save.  All other writes return true (fake success)
185     * so WooCommerce sees "write OK" and does not retry or cascade.
186     *
187     * Hooked on both update_post_metadata and add_post_metadata at priority 0.
188     * The $check parameter is null when no earlier filter has acted; returning
189     * null means "proceed normally", any other value short-circuits the write.
190     *
191     * Also blocks the subsequent _sku write during intercepted classic-editor
192     * saves of a translated PRODUCT post. Does NOT cover the WC_AJAX
193     * variation-save path (save_variations), which is a separate request with
194     * no woocommerce_meta_nonce POST — this is why a translated product's own
195     * Variations tab can still physically write `_sku` onto a translated
196     * variation if someone types a value there directly. That stray row is the
197     * "genuine duplicate" the SKU filters below (pre_has_unique_sku() /
198     * flag_source_sku_conflict()) exist to work around; it does not, on its
199     * own, indicate data corruption.
200     *
201     * @param  mixed  $check     null = proceed; anything else = short-circuit result.
202     * @param  int    $object_id Post ID receiving the meta write.
203     * @param  string $meta_key  Meta key being written.
204     * @return mixed  null to allow; true to fake success without writing.
205     */
206    public static function whitelist_meta_write( mixed $check, int $object_id, string $meta_key ): mixed {
207        if ( ! self::$intercepting ) {
208            return $check;
209        }
210
211        // LF-internal meta — always allow.
212        if ( str_starts_with( $meta_key, '_lf_' ) ) {
213            return $check;
214        }
215
216        // Page template — handle_save_post() corrects the FSE slug here.
217        if ( '_wp_page_template' === $meta_key ) {
218            return $check;
219        }
220
221        // WP housekeeping meta — harmless to allow.
222        if ( '_edit_lock' === $meta_key || '_edit_last' === $meta_key ) {
223            return $check;
224        }
225
226        // Everything else: fake success.  WooCommerce treats the return value of
227        // update_post_meta() / add_post_meta() as truthy = "write succeeded" and
228        // does not retry, validate, or cascade.
229        return true;
230    }

253    /**
254     * Force translated LF products/variations to be treated as SKU-unique,
255     * bypassing WC's own uniqueness check entirely.
256     *
257     * Hooked on `wc_product_pre_has_unique_sku` (WC 9.0+), a genuine short-circuit
258     * filter: a non-null bool returned here IS the final "is unique?" answer
259     * (true = unique), so returning `true` reliably allows the save — unlike the
260     * legacy `wc_product_has_unique_sku` filter (see flag_source_sku_conflict()),
261     * whose polarity is inverted and cannot be used to "allow" a save this way.
262     *
263     * For a translated product/variation, any "duplicate" WC could find is always
264     * the source's own _sku row (or a stray physical row left on the translated
265     * post — see class docblock) — never a genuinely unrelated product. Source
266     * products/variations are left to WC's normal check (return null).
267     *
268     * @param  bool|null $short_circuit  null = no earlier filter decided; proceed normally.
269     * @param  int       $product_id     WooCommerce product/variation ID being validated.
270     * @param  string    $sku            SKU value being checked.
271     * @return bool|null
272     */
273    public static function pre_has_unique_sku( ?bool $short_circuit, int $product_id, string $sku ): ?bool { // phpcs:ignore Generic.CodeAnalysis.UnusedFunctionParameter.Found -- $sku accepted for filter signature compatibility; not needed here.
274        if ( null !== $short_circuit ) {
275            return $short_circuit; // An earlier filter already decided — don't interfere.
276        }
277
278        $post = get_post( $product_id );
279        if ( ! $post || ! in_array( $post->post_type, [ 'product', 'product_variation' ], true ) ) {
280            return null;
281        }
282
283        $lang = (string) get_post_meta( $product_id, '_lf_lang', true );
284        if ( '' === $lang ) {
285            return null; // Not managed by LF.
286        }
287
288        if ( $lang === Router::get_instance()->source_language() ) {
289            return null; // Source product/variation — let WC validate normally.
290        }
291
292        return true; // Translated — always unique from LF's perspective.
293    }

295    /**
296     * Observe genuine SKU conflicts on SOURCE products/variations and flag them
297     * for possible notice suppression at shutdown.
298     *
299     * Hooked on the legacy `wc_product_has_unique_sku` filter. WARNING: this
300     * filter's polarity is the OPPOSITE of "is_unique" despite the shared name —
301     * WC passes $sku_found (true = a duplicate WAS found via direct SQL against
302     * wc_product_meta_lookup), and a truthy value returned from the filter chain
303     * makes the outer wc_product_has_unique_sku() function return false (not
304     * unique). This method never changes that outcome for source products: WC's
305     * own validation is left to stand (the actual save is genuinely blocked when
306     * a real conflict exists), it only records the candidate for the shutdown
307     * notice-suppression pass in suppress_sku_error_before_store().
308     *
309     * Translated products/variations are already resolved upstream by
310     * pre_has_unique_sku() (WC never reaches this filter for them once that
311     * short-circuits), so this method passes them through unchanged as a
312     * defensive fallback only.
313     *
314     * WARNING — do not "fix" this by returning true for translated products.
315     * This class used to make exactly that mistake: the pre-2.4.1
316     * `allow_source_sku_on_translated()` override returned `true` from this
317     * filter for translated products, which WC reads as "yes, duplicate" (see
318     * the polarity warning above) — so translated saves were blocked whenever
319     * a real duplicate row existed, defeating the whole point of the
320     * delegation model. Verified failing with a standalone WC-filter-chain
321     * replica before this fix; see AUDIT-2026-06-06.md follow-up. The correct
322     * fix is pre_has_unique_sku() above, which short-circuits before this
323     * filter is ever reached for translated products/variations.
324     *
325     * @param  bool   $sku_found  true = WC's direct SQL check found a conflicting row.
326     * @param  int    $product_id WooCommerce product ID being validated.
327     * @param  string $sku        SKU value being checked.
328     * @return bool
329     */
330    public static function flag_source_sku_conflict( bool $sku_found, int $product_id, string $sku ): bool {
331        if ( ! $sku_found ) {
332            return $sku_found; // No conflict — nothing to flag.
333        }

335        $post = get_post( $product_id );
