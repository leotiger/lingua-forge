273     *
274     * Covers the three taxonomy groups that determine WC product behaviour:

276     *                     renders a translated variable product as simple.
277     *   - pa_*          — Attribute taxonomies. WC reads attribute term assignments
278     *                     to build variation dropdowns and the attributes table.
279     *   - product_brand — Native WC 10.x brand taxonomy; not content-translated.
280     *

282     * versions. Translated product_cat / product_tag are handled by TaxonomyDelegate
283     * at runtime and are intentionally NOT copied here.
284     *
285     * @param int $source_id      Source (EN) product post ID.
286     * @param int $translated_id  Translated product post ID.
287     */

289
290        if ( ! $source_id || ! $translated_id || $source_id === $translated_id ) {
291            return;
292        }
293
294        // product_type — must match source exactly.
295        $type_terms = wp_get_object_terms( $source_id, 'product_type', [ 'fields' => 'slugs' ] );
296        if ( ! is_wp_error( $type_terms ) && ! empty( $type_terms ) ) {
297            wp_set_object_terms( $translated_id, $type_terms, 'product_type' );
298        }
299
300        // product_brand — native WC 10.x taxonomy; brand identity is language-neutral.
301        if ( taxonomy_exists( 'product_brand' ) ) {
302            $brand_terms = wp_get_object_terms( $source_id, 'product_brand', [ 'fields' => 'ids' ] );
303            if ( ! is_wp_error( $brand_terms ) ) {
304                wp_set_object_terms( $translated_id, $brand_terms, 'product_brand' );
305            }
306        }
307
308        // pa_* attribute taxonomies — term slugs are language-neutral (TermNameFilter
309        // handles display-time translation of term labels separately).
310        $all_taxonomies = get_object_taxonomies( 'product' );
311        foreach ( $all_taxonomies as $taxonomy ) {
312            if ( ! str_starts_with( $taxonomy, 'pa_' ) ) {
313                continue;
314            }
315            $term_ids = wp_get_object_terms( $source_id, $taxonomy, [ 'fields' => 'ids' ] );
316            if ( ! is_wp_error( $term_ids ) ) {
317                wp_set_object_terms( $translated_id, $term_ids, $taxonomy );
318            }
319        }
320
321        // Flush WC product type cache so next wc_get_product() sees the new type.
322        if ( function_exists( 'wc_delete_product_transients' ) ) {
323            wc_delete_product_transients( $translated_id );
324        }
325        \WC_Cache_Helper::invalidate_cache_group( 'product_' . $translated_id );
326    }
