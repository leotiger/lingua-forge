17     * Returns the AI-supported language map filtered to languages active on

20     * linguaforge_languages() (defined in language-router/language-router.php,

22     * that the Language Router recognises as valid for this install — derived

28     *
29     * Sorted by language code (ksort) so the translate-to dropdown lists
30     * languages in a predictable order rather than Translation::get_languages()'s
31     * built-in insertion order.
32     *
33     * @return array<string,string>  e.g. ['en' => 'English', 'es' => 'Spanish']

44    /**
45     * Filter callback — auto-injects any language active on this WordPress
46     * instance (from linguaforge_languages()) that is absent from the built-in
47     * AI language map.
48     *
49     * Registered at priority 5 so user-supplied filters (priority 10) can still
50     * override the derived name if needed.
51     *
52     * English language names are resolved via PHP's Locale class (intl
53     * extension). When intl is unavailable the locale code is uppercased as a
54     * last-resort fallback (e.g. "EU"). The result is good enough for AI prompt
55     * usage in all cases where intl is present (true on virtually every
56     * modern PHP host).
57     *
58     * @param  array<string,string> $languages  Current language map.
59     * @return array<string,string>
60     */
61    public static function inject_instance_languages( array $languages ): array {
62
63        foreach ( linguaforge_languages() as $code ) {
64
65            if ( isset( $languages[ $code ] ) ) {
66                continue;
67            }
68
69            if ( class_exists( 'Locale' ) ) {
70                $name = \Locale::getDisplayLanguage( $code, 'en' );
71                // Locale::getDisplayLanguage() returns the code itself when it
72                // cannot resolve the name — treat that as a failure.
73                if ( $name === '' || $name === $code ) {
74                    $name = strtoupper( $code );
75                }
76            } else {
77                $name = strtoupper( $code );
78            }
79
80            $languages[ $code ] = $name;
81        }
82
83        return $languages;
84    }
