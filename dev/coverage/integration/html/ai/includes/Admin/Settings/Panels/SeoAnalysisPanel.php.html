16 *   landing  — Landing / Short-form. Lower word-count expectations;
17 *              H2 and internal links treated as optional.
18 *

20 *   can be overridden via the profile dropdown in the Settings panel or via
21 *   the `profile` AJAX param.
22 *
23 * ── Sprint 1 metrics (rule-based, zero cost) ──────────────────────────────
24 *   Title            — character length, presence
25 *   Meta description — presence, character length (LF AI meta preferred)
26 *   Word count       — total words in post content
27 *   Reading time     — estimated at 200 wpm
28 *   Headings         — H1 count (should be 1), H2/H3 presence
29 *   Images           — total count, images with/without alt text

361        $lang     = sanitize_key( $_POST['lang']    ?? '' );
362        // phpcs:ignore WordPress.Security.ValidatedSanitizedInput.MissingUnslash
363        $profile_override = sanitize_key( $_POST['profile'] ?? '' );
364
365        if ( ! $post_id ) {
366            wp_send_json_error( [ 'message' => __( 'Invalid post ID.', 'lingua-forge' ) ] );
367        }
368
369        $post = get_post( $post_id );
370
371        if ( ! $post || 'publish' !== $post->post_status ) {
372            wp_send_json_error( [ 'message' => __( 'Post not found or not published.', 'lingua-forge' ) ] );
373        }
374
375        $analyzed_post = $post;
376        $used_lang     = $lang;
377        $used_source   = false;
378
379        // If a specific language is requested, try to find that translation.
380        if ( '' !== $lang ) {
381            $router       = Router::get_instance();
382            $post_lang    = (string) get_post_meta( $post_id, '_lf_lang', true );
383            $source_lang  = $router->context->source_language();
384
385            if ( $post_lang !== $lang ) {
386                $trid = (string) get_post_meta( $post_id, '_lf_trid', true );
387                if ( '' !== $trid ) {
388                    $translations = $router->trid_group->get_translations( $post_id );
389                    if ( isset( $translations[ $lang ] ) ) {
390                        $translated = get_post( $translations[ $lang ] );
391                        if ( $translated ) {
392                            $analyzed_post = $translated;
393                        } else {
394                            $used_source = true;
395                        }
396                    } else {
397                        $used_source = true;
398                    }
399                } elseif ( $post_lang !== $lang && $post_lang !== $source_lang ) {
400                    $used_source = true;
401                }
402            }
403        }
404
405        $profile_key  = self::resolve_profile( get_post_type( $analyzed_post ) ?: 'post', $profile_override );
406        $profile_data = self::profiles()[ $profile_key ];
407
408        $metrics = self::analyze( $analyzed_post, $profile_data );
409
410        wp_send_json_success( [
411            'post_id'     => $analyzed_post->ID,
412            'post_title'  => get_the_title( $analyzed_post ),
413            'lang'        => $used_lang,
414            'used_source' => $used_source,
415            'profile'     => $profile_key,
416            'metrics'     => $metrics,
417            'score'       => self::compute_score( $metrics, $profile_data['weights'] ),
418        ] );
419    }

425    /**
426     * Run all rule-based checks on a post.
427     *
428     * @param  \WP_Post             $post
429     * @param  array<string, mixed> $profile  Profile data from self::profiles(). Defaults to blog profile.
430     * @return array<string, array<string, mixed>>
431     */
432    private static function analyze( \WP_Post $post, array $profile = [] ): array {
433
434        if ( empty( $profile ) ) {
435            $profile = self::profiles()['blog'];
436        }
437
438        $title       = get_the_title( $post );
439        $content     = wp_strip_all_tags( do_blocks( $post->post_content ) );
440        $raw_html    = do_blocks( $post->post_content );
441        $meta_desc   = (string) get_post_meta( $post->ID, '_linguaforge_meta_description', true );
442
443        // Fall back to excerpt then to trimmed content.
444        if ( '' === $meta_desc ) {
445            $meta_desc = '' !== $post->post_excerpt
446                ? wp_strip_all_tags( $post->post_excerpt )
447                : wp_trim_words( $content, 30 );
448        }
449
450        $title_len    = mb_strlen( $title );
451        $meta_len     = mb_strlen( $meta_desc );
452        $words        = self::count_words( $content );
453        $reading_time = (int) ceil( $words / 200 );
454        $images       = self::analyze_images( $raw_html );
455        $links        = self::analyze_links( $raw_html );
456        $has_lf_meta  = '' !== (string) get_post_meta( $post->ID, '_linguaforge_meta_description', true );
457        $h2_as_h1     = (bool) get_option( 'linguaforge_seo_h2_as_h1' );
458
459        // Prefer headings extracted from the full rendered frontend page — this
460        // captures the theme-rendered title tag (H1 or H2) which is not present
461        // in post_content.  Falls back to content-only parsing when the page is
462        // unreachable (local dev, staging with HTTP auth, etc.).
463        $permalink = get_permalink( $post->ID );
464        $headings  = ( $permalink ? self::extract_headings_from_url( $permalink ) : null )
465            ?? self::extract_headings( $raw_html );
466
467        return [
468            'title'            => self::rate_title( $title, $title_len, $profile['title'] ),
469            'meta_description' => self::rate_meta( $meta_desc, $meta_len, $has_lf_meta, $profile['meta'] ),
470            'word_count'       => self::rate_words( $words, $profile['words'] ),
471            'reading_time'     => [
472                'value'   => $reading_time,
473                'display' => sprintf(
474                    /* translators: %d: minutes */
475                    _n( '~%d min read', '~%d min read', $reading_time, 'lingua-forge' ),
476                    $reading_time
477                ),
478                'status'  => 'info',
479            ],
480            'headings'         => self::rate_headings( $headings, $profile['h2_required'], $h2_as_h1 ),
481            'images'           => self::rate_images( $images ),
482            'links'            => self::rate_links( $links, $profile['links_required'] ),
483        ];
484    }

497        $max = $thresholds['max'];

507        }
508        if ( $len > $max ) {
509            return [ 'value' => $title, 'length' => $len, 'status' => 'warn',
510                /* translators: %1$d: number of characters, %2$d: recommended max */

512        }
513        return [ 'value' => $title, 'length' => $len, 'status' => 'ok',
514            /* translators: %d: number of characters */
515            'message' => sprintf( __( '%d chars — good length.', 'lingua-forge' ), $len ) ];
516    }

518    /**
519     * @param  array{min:int,max:int} $thresholds  Meta description character range. Defaults to blog profile.
520     * @return array<string, mixed>
521     */
522    public static function rate_meta( string $meta, int $len, bool $is_lf, array $thresholds = [ 'min' => 120, 'max' => 210 ] ): array {
523
524        $min     = $thresholds['min'];
525        $max     = $thresholds['max'];
526        $ideal   = 160; // Soft ideal upper bound; above this is still ok up to $max.
527
528        $source = $is_lf
529            ? __( 'Using LF AI meta description.', 'lingua-forge' )
530            : __( 'No LF meta description set — using excerpt/content.', 'lingua-forge' );
531
532        if ( $len === 0 ) {
533            return [ 'value' => '', 'length' => 0, 'status' => 'fail',
534                'message' => __( 'No meta description found. Generate one via the AI meta description feature.', 'lingua-forge' ) ];
535        }
536        if ( $len < $min ) {
537            return [ 'value' => $meta, 'length' => $len, 'status' => 'warn',
538                /* translators: %1$d: number of characters, %2$d: ideal min, %3$d: ideal max, %4$s: source note */
539                'message' => sprintf( __( '%1$d chars — too short (aim for %2$d–%3$d). %4$s', 'lingua-forge' ), $len, $min + 20, $ideal, $source ) ];
540        }
541        if ( $len > $max ) {
542            return [ 'value' => $meta, 'length' => $len, 'status' => 'warn',
543                /* translators: %1$d: number of characters, %2$d: hard max, %3$s: source note */
544                'message' => sprintf( __( '%1$d chars — likely truncated in SERPs (keep under %2$d). %3$s', 'lingua-forge' ), $len, $max, $source ) ];
545        }
546        if ( $len > $ideal ) {
547            return [ 'value' => $meta, 'length' => $len, 'status' => 'ok',
548                /* translators: %1$d: number of characters, %2$d: ideal max, %3$d: hard max, %4$s: source note */
549                'message' => sprintf( __( '%1$d chars — good, slightly above the ideal %2$d but within the accepted %3$d-char range. %4$s', 'lingua-forge' ), $len, $ideal, $max, $source ) ];
550        }
551        return [ 'value' => $meta, 'length' => $len, 'status' => 'ok',
552            /* translators: %1$d: number of characters, %2$s: source note */
553            'message' => sprintf( __( '%1$d chars — ideal length. %2$s', 'lingua-forge' ), $len, $source ) ];
554    }

557     * @param  array{fail:int,warn:int} $thresholds  Word count thresholds. Defaults to blog profile.

559     */
560    public static function rate_words( int $count, array $thresholds = [ 'fail' => 100, 'warn' => 300 ] ): array {
561
562        $fail = $thresholds['fail'];
563        $warn = $thresholds['warn'];
564
565        if ( $count < $fail ) {
566            return [ 'value' => $count, 'status' => 'fail',
567                /* translators: %1$d: word count, %2$d: recommended minimum */

569        }
570        if ( $count < $warn ) {
571            return [ 'value' => $count, 'status' => 'warn',
572                /* translators: %1$d: word count, %2$d: recommended minimum */

574        }
575        return [ 'value' => $count, 'status' => 'ok',
576            /* translators: %d: word count */
577            'message' => sprintf( __( '%d words.', 'lingua-forge' ), $count ) ];
578    }
579
580    /**
581     * @param  bool $h2_required  Whether missing H2 is a warn (true) or info (false).
582     * @param  bool $h2_as_h1     When true and the rendered page has no H1 but has at
583     *                            least one H2, promote the first H2 to H1. This covers
584     *                            themes that output the post title inside <h2> rather
585     *                            than <h1>. Because we now fetch the rendered page,
586     *                            both the theme title tag and in-content headings are
587     *                            visible, so the H2 from the title will actually be
588     *                            present when this promotion is needed.
589     * @return array<string, mixed>
590     */
591    public static function rate_headings( array $headings, bool $h2_required = true, bool $h2_as_h1 = false ): array {
592
593        $h1 = $headings['h1'] ?? 0;
594        $h2 = $headings['h2'] ?? 0;
595
596        if ( $h2_as_h1 ) {
597            if ( $h1 === 0 && $h2 > 0 ) {
598                // Rendered page has an H2 but no H1 — theme uses H2 for the title.
599                // Promote the first H2 so it is counted as H1.
600                $h1             = 1;
601                $h2--;
602                $headings['h1'] = $h1;
603                $headings['h2'] = $h2;
604            } elseif ( $h1 === 0 ) {
605                // No H1 or H2 detected.  Most likely the rendered-page fetch failed
606                // (local dev, staging auth) so the theme title tag was invisible.
607                // Credit H1 = 1 as the user explicitly declared this theme pattern.
608                // Use 'ok' so the score is not penalised; add a note for transparency.
609                return array_merge( $headings, [
610                    'status'  => 'ok',
611                    'message' => __( 'H2-as-H1 mode enabled — title counted as H1 equivalent. (Heading tags could not be detected from the rendered page; verify your theme outputs the title correctly.)', 'lingua-forge' ),
612                ] );
613            }
614        }
615
616        if ( $h1 === 0 ) {
617            $status  = 'warn';
618            $message = __( 'No H1 found. Typically the post title is the H1 — verify your theme is outputting it.', 'lingua-forge' );
619        } elseif ( $h1 > 1 ) {
620            $status = 'fail';
621            /* translators: %d: number of H1 tags found */
622            $message = sprintf( __( '%d H1 tags found — a page should have exactly one H1.', 'lingua-forge' ), $h1 );
623        } elseif ( $h2 === 0 ) {
624            $status  = $h2_required ? 'warn' : 'ok';
625            $message = $h2_required
626                ? __( '1 H1 ✓ — no H2 subheadings found. Adding H2s improves structure.', 'lingua-forge' )
627                : __( '1 H1 ✓ — no H2 subheadings (acceptable for this content type).', 'lingua-forge' );
628        } else {
629            $status  = 'ok';
630            $message = sprintf(
631                /* translators: %1$d: number of H2 headings, %2$d: number of H3 headings */
632                __( '1 H1 ✓, %1$d H2, %2$d H3.', 'lingua-forge' ),
633                $h2,
634                $headings['h3'] ?? 0
635            );
636        }
637
638        return array_merge( $headings, [ 'status' => $status, 'message' => $message ] );
639    }
