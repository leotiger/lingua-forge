3 * Class LinguaForge\AI\Admin\Language\LanguageUninstaller
4 *
5 * Removes all Lingua Forge content for a given language code and deletes the
6 * corresponding WordPress locale pack files so the language no longer appears
7 * in the router's active language list.
8 *
9 * Deletion is intentionally synchronous. On large catalogs this may approach
10 * the PHP time limit on shared hosting; §10.2 (Action Scheduler async dispatch)
11 * is the planned future wrapper and will call this class unchanged.
12 *
13 * Two languages are permanently protected:
14 *   • The primary content language  (linguaforge_primary_language option).
15 *   • The WordPress instance locale (get_locale(), first two characters).
16 *
17 * is_protected() must be checked before calling uninstall(). The method itself
18 * also enforces the guard and returns an empty result if called on a protected
19 * language.
20 *
21 * ── Testability ──────────────────────────────────────────────────────────────
22 *
23 * Constructor-injected dependencies:
24 *   • \wpdb        $wpdb    — for collect_post_ids(); mock in unit tests.
25 *   • Router       $router  — for source_language() guard; stub via inject_router().

28 *   • collect_post_ids()   — pure $wpdb->get_col() call.
29 *   • collect_locale_files() — filesystem glob + prefix filter; pass a temp dir.
30 *   • collect_override_files() — i18n-overrides dir glob + suffix filter; pass a temp dir.

275    /**
276     * Uninstall a language: delete all content, remove locale files, flush rules.
277     *
278     * Returns an empty UninstallResult (posts_deleted = 0, empty file lists,
279     * patterns_deleted = 0) if $lang is protected — callers should call
280     * is_protected() first for a clean early-return UX, but this guard prevents
281     * accidental damage if skipped.
282     *
283     * Order of operations:
284     *   1. Guard check.
285     *   2. Collect and delete all posts with _lf_lang = $lang — this covers
286     *      every post-based entity (posts, pages, CPTs, templates, template
287     *      parts, navigation menus; see collect_post_ids() docblock).
288     *   3. If file mods are allowed: collect and delete locale files.
289     *      If not: collect the paths so the caller can surface them in a notice.
290     *   4. Remove the language's CPT Block Pattern translations. Patterns are
291     *      stored in the `linguaforge_pattern_translations` option rather than
292     *      as posts (see PatternDiscovery::save_translation()), so step 2's
293     *      postmeta query never touches them — without this step they would
294     *      survive uninstall indefinitely as orphaned option data.
295     *   5. flush_rewrite_rules() — removes language-prefix rewrite rules that
296     *      referenced the now-deleted locale pack.
297     *
298     * @param  string $lang  Two-character language code.
299     */
300    public function uninstall( string $lang ): UninstallResult {
301
302        $mods_allowed = \wp_is_file_mod_allowed( 'download_language_pack' );
303
304        // Guard — fail safe, no deletions.
305        if ( $this->is_protected( $lang ) ) {
306            return new UninstallResult( 0, [], [], $mods_allowed, 0 );
307        }
308
309        // ── 1. Delete content ─────────────────────────────────────────────────
310        $post_ids      = $this->collect_post_ids( $lang );
311        $posts_deleted = $this->delete_posts( $post_ids );
312
313        // ── 2. Delete locale files ────────────────────────────────────────────
314        // Both WP_LANG_DIR (core/plugin/theme packs) and the i18n-overrides
315        // directory (Loco Translate copies, manual uploads) must be checked —
316        // a language can be active in the router via either source alone.
317        $locale_files  = array_merge(
318            $this->collect_locale_files( $lang ),
319            $this->collect_override_files( $lang )
320        );
321        $files_deleted = [];
322        $files_skipped = [];
323
324        if ( $mods_allowed ) {
325            $file_result   = $this->delete_locale_files( $locale_files );
326            $files_deleted = $file_result['deleted'];
327            $files_skipped = $file_result['failed'];
328        } else {
329            // Surface the paths so the caller can show a manual-deletion notice.
330            $files_skipped = $locale_files;
331        }
332
333        // ── 3. Remove CPT Block Pattern translations ────────────────────────────
334        $patterns_deleted = PatternDiscovery::delete_language( $lang );
335
336        // ── 4. Flush rewrite rules ─────────────────────────────────────────────
337        \flush_rewrite_rules();
338
339        return new UninstallResult( $posts_deleted, $files_deleted, $files_skipped, $mods_allowed, $patterns_deleted );
340    }
