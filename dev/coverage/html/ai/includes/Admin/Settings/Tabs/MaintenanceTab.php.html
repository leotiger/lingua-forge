610    public static function handle_upload_override(): void {
611
612        if (!current_user_can('manage_options')) {
613            wp_die(esc_html__('Permission denied.', 'lingua-forge'), 403);
614        }
615
616        check_admin_referer('linguaforge_upload_override', 'linguaforge_override_nonce');
617
618        $redirect_base = admin_url('options-general.php?page=' . SettingsPage::PAGE_SLUG);
619
620        // No file submitted
621        if (empty($_FILES['linguaforge_mo_file']['name'])) {
622            wp_safe_redirect(add_query_arg('lf_override_error', 'empty', $redirect_base));
623            exit;
624        }
625
626        // phpcs:ignore WordPress.Security.ValidatedSanitizedInput.InputNotSanitized -- $_FILES array is passed directly to wp_handle_upload() which performs its own validation.
627        $file = $_FILES['linguaforge_mo_file'];
628
629        // Validate extension — only .mo files are loaded at runtime
630        $ext = strtolower(pathinfo($file['name'], PATHINFO_EXTENSION));
631        if ($ext !== 'mo') {
632            wp_safe_redirect(add_query_arg('lf_override_error', 'invalid_type', $redirect_base));
633            exit;
634        }
635
636        // Validate upload integrity
637        if ($file['error'] !== UPLOAD_ERR_OK) {
638            wp_safe_redirect(add_query_arg('lf_override_error', 'upload_error', $redirect_base));
639            exit;
640        }
641
642        $dir = self::overrides_dir();
643
644        wp_mkdir_p($dir);
645
646        // Redirect wp_handle_upload() to our override directory and preserve the
647        // exact filename so the {textdomain}-{locale}.mo convention is maintained.
648        $upload_dir_cb = static function ( $dirs ) use ( $dir ) {
649            $dirs['path']   = untrailingslashit( $dir );
650            $dirs['url']    = '';
651            $dirs['subdir'] = '';
652            return $dirs;
653        };
654
655        // Register .mo as an allowed upload type so wp_handle_upload() can run
656        // its MIME-magic check (test_type: true, the default). Without this,
657        // WordPress would reject .mo as an unrecognised type. The check verifies
658        // that the uploaded bytes match application/octet-stream — a PHP file or
659        // image renamed to .mo would produce a different MIME and be rejected.
660        $upload_mimes_cb = static function ( $mimes ) {
661            $mimes['mo'] = 'application/octet-stream';
662            return $mimes;
663        };
664
665        add_filter( 'upload_mimes', $upload_mimes_cb );
666        add_filter( 'upload_dir',   $upload_dir_cb );
667
668        $uploaded = wp_handle_upload(
669            $file,
670            [
671                'test_form'                => false, // nonce already verified via check_admin_referer
672                'unique_filename_callback' => static fn( $d, $n, $e ) => $n, // keep exact name
673            ]
674        );
675
676        remove_filter( 'upload_mimes', $upload_mimes_cb );
677        remove_filter( 'upload_dir',   $upload_dir_cb );
678
679        if ( isset( $uploaded['error'] ) || empty( $uploaded['file'] ) ) {
680            wp_safe_redirect(add_query_arg('lf_override_error', 'move_failed', $redirect_base));
681            exit;
682        }
683
684        // A new .mo override may introduce a previously-unknown locale to
685        // \LinguaForge\Router\Router::languages(), which in turn feeds the rewrite rule
686        // set built on init. Mark the rules dirty so the init-priority-99 hook
687        // in lingua-forge.php picks the flush up on the next request — without
688        // this the new /xx/ URLs return 404 until Settings → Permalinks → Save.
689        update_option( 'linguaforge_flush_rewrite_rules', true, false );
690
691        wp_safe_redirect(add_query_arg('lf_override_uploaded', '1', $redirect_base));
692        exit;
693    }

695    public static function handle_delete_override(): void {
696
697        if (!current_user_can('manage_options')) {
698            wp_die(esc_html__('Permission denied.', 'lingua-forge'), 403);
699        }
700
701        check_admin_referer('linguaforge_delete_override', 'linguaforge_override_nonce');
702
703        $redirect_base = admin_url('options-general.php?page=' . SettingsPage::PAGE_SLUG);
704
705        $filename = sanitize_file_name( wp_unslash( $_POST['linguaforge_override_file'] ?? '' ) );
706
707        // Validate: must be a .mo filename, no path separators
708        if ($filename === '' || strpos($filename, '/') !== false || strpos($filename, '\\') !== false) {
709            wp_safe_redirect(add_query_arg('lf_override_error', 'invalid_file', $redirect_base));
710            exit;
711        }
712
713        if (strtolower(pathinfo($filename, PATHINFO_EXTENSION)) !== 'mo') {
714            wp_safe_redirect(add_query_arg('lf_override_error', 'invalid_type', $redirect_base));
715            exit;
716        }
717
718        $dir     = self::overrides_dir();
719        $real_dir = realpath($dir);
720
721        if ($real_dir === false) {
722            wp_safe_redirect(add_query_arg('lf_override_error', 'invalid_path', $redirect_base));
723            exit;
724        }
725
726        // Delete both .mo and .po for the given base name.
727        $base      = pathinfo($filename, PATHINFO_FILENAME); // strip extension
728        $deleted   = false;
729
730        foreach (['mo', 'po'] as $ext) {
731            $filepath  = $dir . $base . '.' . $ext;
732            $real_file = realpath($filepath);
733
734            // Path-traversal guard: resolved path must still be inside the overrides dir
735            if ($real_file === false || strpos($real_file, $real_dir . DIRECTORY_SEPARATOR) !== 0) {
736                continue;
737            }
738
739            wp_delete_file($filepath);
740            $deleted = true;
741        }
742
743        if (!$deleted) {
744            wp_safe_redirect(add_query_arg('lf_override_error', 'invalid_path', $redirect_base));
745            exit;
746        }
747
748        // Deleting the last .mo for a discovered-only locale removes it from
749        // \LinguaForge\Router\Router::languages(), so the rewrite-rule set must rebuild.
750        // Same mechanism as on upload — defer to init-priority-99 in lingua-forge.php.
751        update_option( 'linguaforge_flush_rewrite_rules', true, false );
752
753        wp_safe_redirect(add_query_arg('lf_override_deleted', '1', $redirect_base));
754        exit;
755    }
