393    /**

395     * parsed block tree.
396     *
397     * @param array<int, array<string, mixed>> $blocks Output of parse_blocks().
398     * @param array<string, string>            $parts  Accumulator: slug → area.
399     */

401        foreach ( $blocks as $block ) {
402            if ( 'core/template-part' === ( $block['blockName'] ?? '' ) ) {

406                    $parts[ $slug ] = $area;
407                }
408            }
409            if ( ! empty( $block['innerBlocks'] ) ) {
410                self::collect_parts_from_blocks( $block['innerBlocks'], $parts );
411            }
412        }

415    /**
416     * Discover which template parts are referenced by the theme's base templates.
417     *
418     * Two-pass approach:
419     *   Pass 1 – collect every core/template-part slug found in the base
420     *            template block markup (area attribute from the block is kept
421     *            only as a last-resort fallback because many themes omit it).
422     *   Pass 2 – resolve the canonical area from the wp_template_part object
423     *            itself (read from the wp_template_part_area taxonomy), so that
424     *            the Site Editor groups the scaffolded part correctly even when
425     *            the block comment carries no area attribute.
426     *
427     * @param string $theme Active theme stylesheet (get_stylesheet()).
428     * @return array<string, string> Slug → area map, sorted by slug.
429     */
430    private static function discover_template_parts( string $theme ): array {
431        // Pass 1 — harvest slugs (and block-level area as a fallback).
432        $raw = [];
433        foreach ( array_keys( self::template_definitions() ) as $base_slug ) {
434            $tpl = get_block_template( $theme . '//' . $base_slug );
435            if ( $tpl && $tpl->content ) {
436                self::collect_parts_from_blocks( parse_blocks( $tpl->content ), $raw );
437            }
438        }
439
440        // Pass 2 — authoritative area from the template part object.
441        $parts = [];
442        foreach ( array_keys( $raw ) as $slug ) {
443            $part           = get_block_template( $theme . '//' . $slug, 'wp_template_part' );
444            $parts[ $slug ] = ( $part && $part->area ) ? $part->area : ( $raw[ $slug ] ?? 'uncategorized' );
445        }
446
447        ksort( $parts );
448        return $parts;
449    }

451    /**

453     *
454     * Modifies the $blocks array in place. Returns true if any replacement
455     * was made (so the caller knows whether to re-serialise the template).

458     * @param string                           $old_slug  Slug to look for.
459     * @param string                           $new_slug  Replacement slug.
460     */

462        $changed = false;
463        foreach ( $blocks as &$block ) {
464            if (
465                'core/template-part' === ( $block['blockName'] ?? '' ) &&
466                ( $block['attrs']['slug'] ?? '' ) === $old_slug
467            ) {

469                $changed = true;

472                if ( self::replace_part_slug_in_blocks( $block['innerBlocks'], $old_slug, $new_slug ) ) {
473                    $changed = true;
474                }
475            }
476        }
477        unset( $block );
478        return $changed;
479    }

744    /**
745     * Expand every wp:pattern block reference inside a content string to its
746     * registered markup, so the FSE translator sees actual translatable text
747     * rather than a bare pattern pointer.
748     *
749     * Resolution order per slug:
750     *   1. WP_Block_Patterns_Registry (theme / plugin PHP-registered patterns
751     *      and theme patterns/ directory entries).
752     *   2. wp_block synced-pattern posts (formerly "reusable blocks") — matched
753     *      by post_name derived from the slug's tail segment.
754     *
755     * If a slug can't be resolved the original <!-- wp:pattern … /--> comment
756     * is left untouched so the AI still sees a valid block structure.
757     *
758     * @param  string $content  Raw block markup that may contain wp:pattern refs.
759     * @return string           Markup with pattern references expanded.
760     */
761    private static function expand_pattern_refs( string $content ): string {
762
763        $registry = \WP_Block_Patterns_Registry::get_instance();
764
765        $expanded = preg_replace_callback(
766            '/<!--\s*wp:pattern\s+(\{[^}]+\})\s*\/-->/i',
767            static function ( array $m ) use ( $registry ): string {
768
769                $attrs = json_decode( $m[1], true );
770                $slug  = isset( $attrs['slug'] ) ? (string) $attrs['slug'] : '';
771
772                if ( $slug === '' ) {
773                    return $m[0];
774                }
775
776                // 1 — PHP-registered / theme-directory pattern.
777                $pattern = $registry->is_registered( $slug )
778                    ? $registry->get_registered( $slug )
779                    : null;
780
781                if ( $pattern && ! empty( $pattern['content'] ) ) {
782                    return (string) $pattern['content'];
783                }
784
785                // 2 — Synced pattern stored as wp_block post.
786                // The post_name is the final path segment of the slug.
787                $name = ltrim( (string) strrchr( $slug, '/' ), '/' );
788                if ( $name !== '' ) {
789                    $posts = get_posts( [
790                        'post_type'      => 'wp_block',
791                        'name'           => $name,
792                        'posts_per_page' => 1,
793                        'post_status'    => 'publish',
794                    ] );
795                    if ( ! empty( $posts ) ) {
796                        return (string) $posts[0]->post_content;
797                    }
798                }
799
800                return $m[0]; // Unresolvable — leave the reference intact.
801            },
802            $content
803        );
804
805        return is_string( $expanded ) ? $expanded : $content;
806    }
