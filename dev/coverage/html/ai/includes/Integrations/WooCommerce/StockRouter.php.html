2/**
3 * Class LinguaForge\AI\Integrations\WooCommerce\StockRouter
4 *
5 * Routes WooCommerce stock WRITES to the source-language product.
6 *
7 * With the shared-stock delegation model, MetaDelegate transparently redirects
8 * all operational meta READS (including _stock, _stock_status, etc.) to the
9 * source product. However, WooCommerce stock WRITES — triggered by purchases,
10 * manual stock adjustments, and WC's stock management UI — target the product
11 * ID they were given. If a customer purchases a translated product, the order
12 * item carries the translated product ID, and WooCommerce will attempt to write
13 * the new stock level to that translated product post.
14 *
15 * This class intercepts `update_post_metadata` for stock keys on translated
16 * product posts and redirects the write to the source product instead. Returning
17 * `true` from the filter short-circuits WordPress's own DB write, ensuring the
18 * translated post remains meta-free for the stock keys, and the source product
19 * is the single source of truth.
20 *
21 * Stock keys covered: _stock, _stock_qty, _stock_status, _manage_stock,
22 * _backorders, _sold_individually.
23 *
24 * @package LinguaForge\AI\Integrations\WooCommerce
25 * @since   2.0.0
26 */
27

123     * @param  mixed  $meta_value Value to write.

125     * @return mixed

127    private static function maybe_route( mixed $check, int $object_id, string $meta_key, mixed $meta_value, bool $is_add ): mixed {
128
129        // ── 1. Key guard ───────────────────────────────────────────────────────
130        if ( ! in_array( $meta_key, self::STOCK_WRITE_KEYS, true ) ) {
131            return $check;
132        }
133
134        // ── 2. Reentrancy guard ────────────────────────────────────────────────
135        $guard = $object_id . ':' . $meta_key;
136        if ( isset( self::$routing[ $guard ] ) ) {
137            return $check;
138        }
139
140        // ── 3. Post type guard ─────────────────────────────────────────────────
141        $post = get_post( $object_id );
142        if ( ! $post ) {
143            return $check;
144        }
145
146        $delegate_types = (array) apply_filters( 'linguaforge_wc_delegate_post_types', [ 'product' ] );
147        if ( ! in_array( $post->post_type, $delegate_types, true ) ) {
148            return $check;
149        }
150
151        // ── 4. Language guard — only route for translated (non-source) posts ───
152        $lang = (string) get_post_meta( $object_id, '_lf_lang', true );
153        if ( '' === $lang ) {
154            return $check;
155        }
156
157        $source_lang = Router::get_instance()->source_language();
158        if ( $lang === $source_lang ) {
159            return $check; // Already writing to the source — no routing needed.
160        }
161
162        // ── 5. Resolve source product ID ──────────────────────────────────────
163        $source_id = MetaDelegate::get_source_id_for( $object_id );
164        if ( ! $source_id || $source_id === $object_id ) {
165            return $check; // Fail safe — let the write proceed on the translated post.
166        }
167
168        // ── 6. Route write to source product ──────────────────────────────────
169        self::$routing[ $guard ] = true;
170
171        if ( $is_add ) {
172            add_post_meta( $source_id, $meta_key, $meta_value );
173        } else {
174            update_post_meta( $source_id, $meta_key, $meta_value );
175        }
176
177        unset( self::$routing[ $guard ] );
178
179        // Return true: "pretend the write succeeded on the translated post".
180        // The actual write went to the source. The translated post remains
181        // meta-free for this key, so MetaDelegate continues to serve the
182        // source product's value on all subsequent reads.
183        return true;
184    }
